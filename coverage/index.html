type Agent struct {
LLM   llm.Client
Index *rag.Index
Tools *tools.Registry
}

type Response struct {
Type    string        `json:"type"`
Text    string        `json:"text,omitempty"`
Action  *tools.Result `json:"action,omitempty"`
Sources []Source      `json:"sources,omitempty"`
}

func (a *Agent) Ask(ctx context.Context, user auth.User, question string) (Response, error) {
hits := a.Index.Search(question, user.TeamID, 4)
srcs := make([]Source, 0, len(hits))
for _, h := range hits {
srcs = append(srcs, Source{ID: h.Chunk.ID, Source: h.Chunk.Source, Stage: h.Chunk.Stage})
}
if len(srcs) == 0 {
return Response{Type: "refusal", Text: "Не нашёл в доступных источниках. Уточните у наставника."}, nil
}

msgs := []llm.Message{
{Role: "system", Content: SystemPrompt},
{Role: "user", Content: UserPrompt(question, srcs, string(user.Role))},
}
raw, err := a.LLM.Complete(ctx, msgs)
if err != nil {
return Response{}, err
}

var out AgentOutput
if err := json.Unmarshal([]byte(raw), &out); err != nil {
return Response{}, errors.New("invalid llm json")
}

if out.Type == "tool_call" {
t, ok := a.Tools.Get(out.Tool)
if !ok {
return Response{Type: "refusal", Text: "Инструмент недоступен"}, nil
}
res, err := t.Run(ctx, user, out.Args)
if err != nil {
return Response{Type: "refusal", Text: "Действие недоступно: " + err.Error()}, nil
}
return Response{Type: "action", Action: &res, Sources: srcs}, nil
}
if out.Type == "refusal" {
return Response{Type: "refusal", Text: out.Text}, nil
}
return Response{Type: "answer", Text: out.Text, Sources: srcs}, nil
}

const SystemPrompt = `Ты — AI-buddy, помощник стажёра Авито (Backend-команда).
Правила:
1. Отвечай ТОЛЬКО по фрагментам из блока CONTEXT. Если ответа там нет — верни {"type":"refusal","text":"Не нашёл в доступных источниках, предлагаю спросить наставника"}.
2. Никогда не раскрывай информацию, недоступную роли пользователя.
3. Если нужно выполнить действие — верни {"type":"tool_call","tool":"<name>","args":{...}}.
4. Если это ответ — верни {"type":"answer","text":"...","sources":[{"id":"...","source":"...","stage":"..."}]}.
5. Отвечай кратко, по-русски, со ссылкой на источник.
Формат ответа — строго JSON, без markdown.`

func UserPrompt(question string, hits []Source, role string) string {
s := "ROLE: " + role + "\nQUESTION: " + question + "\nCONTEXT: "
for _, h := range hits {
s += "[" + h.ID + " | " + h.Source + " | " + h.Stage + "] "
}
return s
}

return true
case "absence.create":

return u.Role == RoleIntern || u.Role == RoleBuddy || u.Role == RoleLead
case "absence.read.all":
return u.Role == RoleLead || u.Role == RoleHRBP
default:
return false
}
}

func LoadDotEnv(path string) {
f, err := os.Open(path)
if err != nil {
return
}
defer f.Close()
sc := bufio.NewScanner(f)
for sc.Scan() {
line := strings.TrimSpace(sc.Text())
if line == "" || strings.HasPrefix(line, "#") {
continue
}
parts := strings.SplitN(line, "=", 2)
if len(parts) != 2 {
continue
}
k := strings.TrimSpace(parts[0])
v := strings.TrimSpace(parts[1])
if os.Getenv(k) == "" {
_ = os.Setenv(k, v)
}
}
}

package rag

import (
"regexp"
"strings"
)

type Chunk struct {
ID     string
Source string
Stage  string
TeamID string
Text   string
}

var headerRe = regexp.MustCompile(`^#+\s+(.*)$`)

func ChunkMarkdown(source, content string) []Chunk {
lines := strings.Split(content, "\n")
var chunks []Chunk
var cur Chunk
var buf []string

flush := func() {
text := strings.TrimSpace(strings.Join(buf, "\n"))
if text == "" {
return
}
cur.Text = text
cur.Source = source
cur.ID = source + "#" + cur.Stage + "#" + itoa(len(chunks))
chunks = append(chunks, cur)
buf = buf[:0]
}

for _, ln := range lines {
if m := headerRe.FindStringSubmatch(ln); m != nil {
flush()
cur = Chunk{Stage: m[1]}
continue
}
if strings.HasPrefix(ln, "<!--") && strings.Contains(ln, "stage:") {
cur.Stage = extract(ln, "stage")
cur.TeamID = extract(ln, "team")
continue
}
buf = append(buf, ln)
}
flush()
return chunks
}

func extract(s, key string) string {
re := regexp.MustCompile(key + `:\s*([^\s]+)`)
if m := re.FindStringSubmatch(s); len(m) == 2 {
return m[1]
}
return ""
}

func itoa(i int) string {
if i == 0 {
return "0"
}
var b []byte
for i > 0 {
b = append([]byte{byte('0' + i%10)}, b...)
i /= 10
}
return string(b)
}

package rag

import (
"math"
"sort"
"strings"
)

type Index struct {
chunks []Chunk
df     map[string]int
tf     []map[string]int
lens   []int
}

func tokenize(s string) []string {
s = strings.ToLower(s)
return strings.FieldsFunc(s, func(r rune) bool {
return !(r >= 'a' && r <= 'z') && !(r >= 'а' && r <= 'я') && !(r >= '0' && r <= '9')
})
}

func NewIndex(chunks []Chunk) *Index {
idx := &Index{chunks: chunks, df: map[string]int{}}
for _, c := range chunks {
tf := map[string]int{}
toks := tokenize(c.Text + " " + c.Stage)
for _, t := range toks {
tf[t]++
}
for t := range tf {
idx.df[t]++
}
idx.tf = append(idx.tf, tf)
idx.lens = append(idx.lens, len(toks))
}
return idx
}

type Hit struct {
Chunk Chunk
Score float64
}

func prefixMatch(a, b string) bool {
if a == b {
return true
}
n := 5
if len(a) < n {
n = len(a)
}
if len(b) < n {
n = len(b)
}
if n < 3 {
return false
}
return a[:n] == b[:n]
}

func (idx *Index) Search(q string, teamID string, k int) []Hit {
qt := tokenize(q)
n := float64(len(idx.chunks))
avg := 0.0
for _, l := range idx.lens {
avg += float64(l)
}
if n > 0 {
avg /= n
}
if avg == 0 {
avg = 1
}

var hits []Hit
for i, c := range idx.chunks {
if teamID != "" && c.TeamID != "" && c.TeamID != teamID {
continue
}
score := 0.0
for _, t := range qt {
for ct, f := range idx.tf[i] {
if prefixMatch(t, ct) {
df := float64(idx.df[ct])
idf := math.Log(1 + (n-df+0.5)/(df+0.5))
score += idf * (float64(f) * 2.2) / (float64(f) + 1.2*(0.25+0.75*float64(idx.lens[i])/avg))
}
}
}
if score > 0 {
hits = append(hits, Hit{Chunk: c, Score: score})
}
}
sort.Slice(hits, func(i, j int) bool { return hits[i].Score > hits[j].Score })
if len(hits) > k {
hits = hits[:k]
}
return hits
}

func LoadDir(dir string) ([]Chunk, error) {
var all []Chunk
entries, err := os.ReadDir(dir)
if err != nil {
return nil, err
}
for _, e := range entries {
if e.IsDir() || filepath.Ext(e.Name()) != ".md" {
continue
}
b, err := os.ReadFile(filepath.Join(dir, e.Name()))
if err != nil {
return nil, err
}
all = append(all, ChunkMarkdown(e.Name(), string(b))...)
}
return all, nil
}

import (

"time"
)

type Absence struct {
ID        string    `json:"id"`
UserID    string    `json:"user_id"`
DateFrom  string    `json:"date_from"`
DateTo    string    `json:"date_to"`
Reason    string    `json:"reason"`
Status    string    `json:"status"`
CreatedAt time.Time `json:"created_at"`
}

type ChecklistItem struct {
ID    string `json:"id"`
Title string `json:"title"`
Stage string `json:"stage"`
Done  bool   `json:"done"`
Owner string `json:"owner"`
}

type Store struct {
mu        sync.RWMutex
absences  map[string]Absence
checklist map[string][]ChecklistItem
}

func New() *Store {
return &Store{
absences:  map[string]Absence{},
checklist: map[string][]ChecklistItem{},
}
}

func (s *Store) CreateAbsence(a Absence) Absence {
s.mu.Lock()
defer s.mu.Unlock()
s.absences[a.ID] = a
return a
}

func (s *Store) GetAbsence(id string) (Absence, bool) {
s.mu.RLock()
defer s.mu.RUnlock()
a, ok := s.absences[id]
return a, ok
}

func (s *Store) SeedChecklist(userID string, items []ChecklistItem) {
s.mu.Lock()
defer s.mu.Unlock()
s.checklist[userID] = items
}

func (s *Store) Checklist(userID string) []ChecklistItem {
s.mu.RLock()
defer s.mu.RUnlock()
return s.checklist[userID]
}

func (s *Store) MarkDone(userID, itemID string) bool {
s.mu.Lock()
defer s.mu.Unlock()
items := s.checklist[userID]
for i := range items {
if items[i].ID == itemID {
items[i].Done = true
s.checklist[userID] = items
return true
}
}
return false
}

type Result struct {

Message string         `json:"message"`
Data    map[string]any `json:"data,omitempty"`
}

type Tool interface {
Name() string
RequiredPermission() string
Run(ctx context.Context, user auth.User, args map[string]string) (Result, error)
}

type Registry struct {
tools map[string]Tool
}

func NewRegistry() *Registry { return &Registry{tools: map[string]Tool{}} }

func (r *Registry) Register(t Tool) { r.tools[t.Name()] = t }

func (r *Registry) Get(name string) (Tool, bool) {
t, ok := r.tools[name]
return t, ok
}
