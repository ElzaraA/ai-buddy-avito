var shutdownTimeout = 5 * time.Second

func main() {
        exit(serve())
}

// serve запускает сервер до SIGINT/SIGTERM и возвращает код выхода процесса.
func serve() int {
        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()
        if err := run(ctx); err != nil {
                slog.Error("server stopped", "err", err)
                return 1
        }
        return 0
}

// run поднимает сервер и останавливает его корректно (graceful shutdown) при отмене ctx.
func run(parent context.Context) error {
        config.LoadDotEnv(".env")
        cfg := config.Load()

        app, err := api.Build(cfg, cfg.KnowledgeDir)
        if err != nil {
                return err
        }

        // Состояние сохраняется на диск раз в 2 с (если менялось) и ещё раз при любой остановке.
        ctx, cancel := context.WithCancel(parent)
        var bg sync.WaitGroup
        defer bg.Wait() // выполняется после cancel(): ждём финального сохранения
        defer cancel()
        if app.Persister != nil {
                bg.Add(1)
                go func() {
                        defer bg.Done()
                        app.Persister.Run(ctx, 2*time.Second)
                }()
        }

        mode := "mock"
        if !cfg.LLMMock {
                mode = "real:" + cfg.LLMModel
        }
        slog.Info("starting", "addr", cfg.Addr, "llm", mode, "state_file", cfg.StateFile, "rate_limit_per_min", cfg.RateLimitPerMin)

        srv := &http.Server{
                Addr:              cfg.Addr,
                Handler:           app.Handler,
                ReadHeaderTimeout: 5 * time.Second,
                IdleTimeout:       120 * time.Second,
                // WriteTimeout не задаём: он оборвёт SSE-поток.
        }
        errCh := make(chan error, 1)
        go func() { errCh <- srv.ListenAndServe() }()

        select {
        case err := <-errCh:
                if errors.Is(err, http.ErrServerClosed) {
                        return nil
                }
                return err
        case <-ctx.Done():
                shutdownCtx, cancelShutdown := context.WithTimeout(context.Background(), shutdownTimeout)
                defer cancelShutdown()
                slog.Info("shutting down")
                // SSE-потоки живут долго: Shutdown ждёт их до таймаута, затем закрываем принудительно.
                if err := srv.Shutdown(shutdownCtx); err != nil {
                        return srv.Close()
                }
                return nil
        }
}

// contacts — к кому направлять пользователя. Стажёра — к наставнику его команды;

type contacts struct {
        auth.Contacts
        foreign []string // фамилии наставников и руководителей других команд
        staff   bool     // наставник или руководитель

func contactsFor(u auth.User) contacts {
        return contacts{
                Contacts: auth.TeamContacts(u.TeamID),
                foreign:  auth.ForeignContactStems(u.TeamID),
                staff:    u.Role == auth.RoleBuddy || u.Role == auth.RoleLead,
        }
}

// mentionsForeign — в тексте назван наставник или руководитель чужой команды.
func (c contacts) mentionsForeign(text string) bool {
        for _, s := range c.foreign {
                if strings.Contains(text, s) {
                        return true
                }
        }
        return false
}

// withHelp приводит текст отказа модели к контактам пользователя: чужое имя или совет
// «обратиться к наставнику» для наставника/руководителя заменяются, отсутствующий совет дописывается.
func (c contacts) withHelp(text string) string {
        low := strings.ToLower(text)
        if c.mentionsForeign(text) || (c.staff && strings.Contains(low, "наставник")) {
                return "Не могу помочь с этим вопросом. " + c.help()
        }
        if strings.Contains(low, "наставник") || strings.Contains(text, "HR BP") {
                return text
        }
        text = strings.TrimSpace(text)
        if !strings.HasSuffix(text, ".") && !strings.HasSuffix(text, "!") && !strings.HasSuffix(text, "?") {
                text += "."
        }
        return text + " " + c.help()
}

type Agent struct {
        LLM llm.Client
        // Fallback — резервная модель (детерминированный mock): используется, если основная недоступна.
        // Ответ помечается degraded=true, сервис продолжает работать.
        Fallback llm.Client
        Index    *rag.Index
        Tools    *tools.Registry
        Now      func() time.Time
        // OnFallback вызывается при переходе на резервную модель (метрика).
        OnFallback func(err error)
}

type Response struct {
        Type     string            `json:"type"`
        Text     string            `json:"text,omitempty"`
        Action   string            `json:"action,omitempty"`
        Args     map[string]string `json:"args,omitempty"`
        Result   *tools.Result     `json:"result,omitempty"`
        Tool     string            `json:"tool,omitempty"` // имя выполненного инструмента (для type=action)
        Sources  []Source          `json:"sources,omitempty"`
        Buddy    string            `json:"buddy_name,omitempty"`
        Manager  string            `json:"manager_name,omitempty"`
        Degraded bool              `json:"degraded,omitempty"` // ответ дан резервной моделью
        // NoSource — отказ из-за отсутствия ответа в базе (сигнал «пробел в базе знаний»).
        NoSource bool `json:"-"`
}

func (a *Agent) now() time.Time {
        if a.Now != nil {
                return a.Now()
        }
        return time.Now()
}

// Ask отвечает на одиночный вопрос без истории диалога.
func (a *Agent) Ask(ctx context.Context, user auth.User, question string) (Response, error) {
        return a.AskWithHistory(ctx, user, question, nil)
}

// complete вызывает основную модель, а при её сбое — резервную.
func (a *Agent) complete(ctx context.Context, msgs []llm.Message) (raw string, degraded bool, err error) {
        raw, err = a.LLM.Complete(ctx, msgs)
        if err == nil {
                return raw, false, nil
        }
        if a.Fallback == nil || errors.Is(ctx.Err(), context.Canceled) {
                return "", false, err
        }
        if a.OnFallback != nil {
                a.OnFallback(err)
        }
        // Резервной модели даём свой короткий таймаут: основной контекст мог уже истечь.
        fctx, cancel := context.WithTimeout(context.WithoutCancel(ctx), 5*time.Second)
        defer cancel()
        raw, ferr := a.Fallback.Complete(fctx, msgs)
        if ferr != nil {
                return "", false, err
        }
        return raw, true, nil
}

// AskWithHistory отвечает на вопрос с учётом последних реплик диалога (history — в хронологическом порядке).
func (a *Agent) AskWithHistory(ctx context.Context, user auth.User, question string, history []llm.Message) (Response, error) {
        ctx, cancel := context.WithTimeout(ctx, 60*time.Second)
        defer cancel()

        c := contactsFor(user)
        // «Мне плохо», «я не справляюсь» — не вопрос к базе знаний: поддерживаем и предлагаем живого человека.
        if isEmotional(question) {
                return Response{Type: "answer", Text: c.supportText(), Buddy: c.Buddy}, nil
        }
        // Приветствие, «спасибо», «ок», «отмени» — не вопрос к базе знаний: отвечаем сразу, без LLM и без отказа.
        if reply := smallTalk(question); reply != "" {
                return Response{Type: "answer", Text: reply, Buddy: c.Buddy}, nil
        }

        hits := a.Index.Search(question, user.TeamID, string(user.Role), 4)
        srcs := make([]Source, 0, len(hits))
        for _, h := range hits {
                srcs = append(srcs, Source{ID: h.Chunk.ID, Source: h.Chunk.Source, Stage: h.Chunk.Stage, Title: h.Chunk.Title})
        }
        // Ответ есть, но в материале для другой аудитории (например, «только для наставников»): это не пробел базы.
        restricted := len(hits) == 0 && a.Index.Restricted(question, user.TeamID, string(user.Role))

        // Действия (отгул, доступ, чек-лист) не зависят от базы знаний, поэтому LLM вызываем всегда,
        // а инвариант «нет источника — нет ответа» проверяем ниже для текстовых ответов.
        if len(history) > maxHistory {
                history = history[len(history)-maxHistory:]
        }
        msgs := make([]llm.Message, 0, len(history)+2)
        msgs = append(msgs, llm.Message{Role: "system", Content: SystemPrompt})
        msgs = append(msgs, history...)
        msgs = append(msgs, llm.Message{Role: "user", Content: UserPrompt(question, hits, string(user.Role))})

        raw, degraded, err := a.complete(ctx, msgs)
        if err != nil {
                return Response{}, err
        }
        resp := a.interpret(ctx, user, question, raw, srcs, restricted)
        resp.Degraded = degraded
        return resp, nil
}

// restrictedRefusal — материал по вопросу есть, но доступен наставникам и руководителям.
func (c contacts) restrictedRefusal() Response {
        if c.staff {
                return c.refusal("Этот материал предназначен для другой роли в команде. " + c.help())
        }
        return c.refusal("Этот материал доступен наставникам и руководителям команды. " + c.help())
}

// interpret превращает вывод модели в ответ, применяя инварианты из specs/agent/spec.md.
// restricted — по вопросу нашёлся только материал для другой аудитории (не пробел базы знаний).
func (a *Agent) interpret(ctx context.Context, user auth.User, question, raw string, srcs []Source, restricted bool) Response {
        c := contactsFor(user)
        raw = stripCodeFence(raw)
        var out AgentOutput
        if err := json.Unmarshal([]byte(raw), &out); err != nil {
                // Модель ответила не JSON — показываем как текст, но только если есть источники.
                out = AgentOutput{Type: "answer", Text: raw}
        }

        switch out.Type {
        case "tool_call":
                t, ok := a.Tools.Get(out.Tool)
                if !ok {
                        return c.refusal("Этот инструмент мне недоступен. " + c.help())
                }
                // Инвариант 2: право проверяется до любого действия, в том числе до подготовки черновика.
                if !auth.Can(user, t.RequiredPermission()) {
                        return c.refusal("У вашей роли нет доступа к этому действию. " + c.help())
                }
                // Инвариант 4: мутирующие инструменты не выполняются сразу — только черновик.
                if t.RequiresConfirmation() {
                        return a.draftFor(user, question, out.Tool, out.Args, srcs)
                }
                res, err := tools.Run(ctx, t, user, out.Args)
                if err != nil {
                        return c.refusal("Действие сейчас недоступно. " + c.help())
                }
                return Response{Type: "action", Result: &res, Tool: t.Name(), Sources: srcs, Buddy: c.Buddy, Manager: c.Manager}

        case "draft":
                tool := draftTool(out.Action)
                if t, ok := a.Tools.Get(tool); ok && !auth.Can(user, t.RequiredPermission()) {
                        return c.refusal("У вашей роли нет доступа к этому действию. " + c.help())
                }
                return a.draftFor(user, question, tool, out.Args, srcs)

        case "refusal":
                if restricted {
                        return c.restrictedRefusal()
                }
                text := out.Text
                // Инвариант 1 действует и для отказа: без источников текст модели не показываем дословно,
                // иначе выдуманный факт («Зарплата стажёра 120 000 ₽») просочится под видом отказа.
                switch {
                case len(srcs) == 0 && looksLikeInjection(question):
                        text = "Не могу выполнить это действие. " + c.help()
                case text == "" || (len(srcs) == 0 && !safeRefusal(text, c)):
                        text = c.refusalText()
                default:
                        text = c.withHelp(text)
                }
                r := c.refusal(text)
                r.NoSource = len(srcs) == 0 && isKnowledgeGap(question)
                return r
        }
        // Инвариант 1: нет источника — нет ответа (защита от галлюцинаций реальной LLM).
        if len(srcs) == 0 || strings.TrimSpace(out.Text) == "" {
                if restricted {
                        return c.restrictedRefusal()
                }
                r := c.refusal(c.refusalText())
                r.NoSource = isKnowledgeGap(question)
                return r
        }
        return Response{Type: "answer", Text: out.Text, Sources: srcs}
}

// DraftFor строит черновик действия для инструмента (используется и агентом, и MCP-эндпоинтом).

// draftFor строит черновик действия; само действие создаётся только после POST /actions/{id}/confirm.
func (a *Agent) draftFor(u auth.User, question, tool string, args map[string]string, srcs []Source) Response {
        c := contactsFor(u)
        switch tool {
        case "absence_create_request":
                return a.absenceDraft(c, question, args, titled(srcs, "отгул"))
        case "access_request":
                res := store.NormalizeResource(args["resource"])
                // Система не названа или не из списка — честно говорим, на что можно оформить заявку,
                // и черновик не создаём (раньше неизвестная система молча становилась Jira).
                if res == "" {
                        return c.refusal("К какой системе нужен доступ? Заявку можно оформить только на " + resourceList +
                                " — напишите, например, «Запроси доступ в GitLab».")
                }
                if !store.ValidResource(res) {
                        return c.refusal("Заявку на доступ к " + foreignResource(res) + " оформить не могу: заявку можно оформить только на " +
                                resourceList + ". " + c.otherSystems())
                }
                reason := args["reason"]
                if reason == "" {
                        reason = "onboarding"
                }
                return Response{
                        Type:    "draft",
                        Action:  store.ActionAccess,
                        Args:    map[string]string{"resource": res, "reason": reason},
                        Text:    "Подготовил заявку на доступ к " + displayResource(res) + ". Проверьте и подтвердите отправку — без подтверждения заявка не уйдёт.",
                        Sources: titled(srcs, "доступ"),
                        Buddy:   c.Buddy,
                        Manager: c.Manager,
                }
        }
        return c.refusal("Не знаю такого действия. " + c.help())
}

// foreignResource — имя системы не из списка для текста ответа: короткое слово показываем
// с заглавной буквы («confluence» → «Confluence»), всё остальное (разметку, длинные строки) — не повторяем.
func foreignResource(r string) string {
        rs := []rune(r)
        if len(rs) == 0 || len(rs) > 30 {
                return "этой системе"
        }
        for _, x := range rs {
                if !unicode.IsLetter(x) && !unicode.IsDigit(x) && x != '-' && x != ' ' && x != '.' {
                        return "этой системе"
                }
        }
        return string(unicode.ToUpper(rs[0])) + string(rs[1:])
}

// titled оставляет у черновика только источники по его теме: «Хочу отгул послезавтра»
// не должен прикладывать код-ревью только потому, что «после» совпало с «после апрува».
func titled(srcs []Source, topic string) []Source {
        var out []Source
        for _, s := range srcs {
                if strings.Contains(strings.ToLower(s.Title), topic) {
                        out = append(out, s)
                }
        }
        return out
}

func displayResource(r string) string {
        switch r {
        case "jira":
                return "Jira"
        case "gitlab":
                return "GitLab"
        case "grafana":
                return "Grafana"
        case "git":
                return "Git"
        }
        return r
}

func (a *Agent) absenceDraft(c contacts, question string, args map[string]string, srcs []Source) Response {
        from, to := args["date_from"], args["date_to"]
        if from == "" {
                from, to = args["dateFrom"], args["dateTo"]
        }
        // Даты от модели принимаем только корректные; иначе разбираем вопрос сами.
        if f, t := ParseDates(from + " " + to); f != "" {
                from, to = f, t
        } else {
                from, to = ParseDatesAt(question, a.now())
        }
        if from == "" {
                if d, ok := ParseWeekday(question, a.now()); ok {
                        from, to = d, d
                }
        }
        base := Response{
                Type:    "draft",
                Action:  store.ActionAbsence,
                Sources: srcs,
                Buddy:   c.Buddy,
                Manager: c.Manager,
        }
        if from == "" {
                base.Args = map[string]string{"reason": "personal"}
                example := HumanDate(a.now().AddDate(0, 0, 7))
                base.Text = "На какую дату оформить отгул? Напишите, например, «" + example + "», «20.11» или «в пятницу». Согласует руководитель — " + c.Manager + "."
                if bad := InvalidISODate(question); bad != "" {
                        base.Text = "Даты " + bad + " не существует. " + base.Text
                }
                return base
        }
        if from < a.now().Format("2006-01-02") {
                base.Args = map[string]string{"reason": "personal"}
                base.Text = "Дата " + from + " уже прошла. На какую будущую дату оформить отгул?"
                return base
        }
        // Те же правила, что у POST /actions: отгул не длиннее 30 дней.
        if err := store.ValidateAbsenceDates(from, to, a.now()); err != nil {
                base.Args = map[string]string{"reason": "personal"}
                base.Text = "Отгул можно оформить не больше чем на 30 дней подряд. Для более долгого отсутствия обратитесь к руководителю — " + c.Manager + ". На какие даты оформить отгул?"
                return base
        }
        base.Args = map[string]string{"date_from": from, "date_to": to, "reason": "personal"}
        base.Text = "Подготовил черновик заявки на отгул. Её согласует руководитель — " + c.Manager + ". "
        if !c.staff {
                // Наставнику и руководителю не советуем «предупредить наставника» — себя или подчинённого.
                base.Args["suggest_notify_buddy"] = "true"
                base.Text += "Предупредите наставника " + c.BuddyAccusative + " — сам писать я не буду. "
        }
        base.Text += "Проверьте даты и подтвердите."
        return base
}

func stripCodeFence(s string) string {
        s = strings.TrimSpace(s)
        if strings.HasPrefix(s, "```") {
                if i := strings.Index(s, "\n"); i >= 0 {
                        s = s[i+1:]
                }
                s = strings.TrimSuffix(s, "```")
                s = strings.TrimSpace(s)
        }
        return s
}

        dotDateRe = regexp.MustCompile(`(?:^|[^\d.])(\d{1,2})\.(\d{1,2})(?:\.(\d{4}|\d{2}))?(?:$|[^\d.]|\.(?:$|\s))`)
        // monthDateRe — «20 ноября», «20-22 ноября», «с 20 по 22 ноября 2026».
        monthDateRe = regexp.MustCompile(`(?:^|[^\d])(\d{1,2})(?:\s*(?:-|–|—|по|до)\s*(\d{1,2}))?\s+(январ|феврал|март|апрел|ма[йя]|июн|июл|август|сентябр|октябр|ноябр|декабр)[а-я]*(?:\s+(\d{4}))?`)
)

// monthStems — основы названий месяцев в порядке месяцев.
var monthStems = []string{"январ", "феврал", "март", "апрел", "ма", "июн", "июл", "август", "сентябр", "октябр", "ноябр", "декабр"}

// monthGenitive — «20 ноября» для подсказок пользователю.
var monthGenitive = []string{"января", "февраля", "марта", "апреля", "мая", "июня", "июля", "августа", "сентября", "октября", "ноября", "декабря"}

// HumanDate — дата в виде «20 ноября».
func HumanDate(t time.Time) string {
        return strconv.Itoa(t.Day()) + " " + monthGenitive[t.Month()-1]
}

// ParseDates вытаскивает корректные ISO-даты из запроса. Если дат нет — возвращает пусто.
func ParseDates(q string) (string, string) {
        var valid []string
        for _, m := range dateRe.FindAllString(q, -1) {
                if _, err := time.Parse("2006-01-02", m); err == nil {
                        valid = append(valid, m)
                }
        }
        return firstTwo(valid)
}

// ParseDatesAt понимает ISO-даты, «20.11[.2026]», «20 ноября» и диапазоны «с 20 по 22 ноября».
// Без года берётся ближайшая будущая дата относительно today.
func ParseDatesAt(q string, today time.Time) (string, string) {
        if f, t := ParseDates(q); f != "" {
                return f, t
        }
        q = strings.ToLower(q)
        type found struct {
                pos  int
                date string
        }
        var all []found
        add := func(pos int, day, month int, year string) {
                if d, ok := makeDate(day, month, year, today); ok {
                        all = append(all, found{pos, d})
                }
        }
        for _, m := range monthDateRe.FindAllStringSubmatchIndex(q, -1) {
                month := monthIndex(q[m[6]:m[7]])
                year := ""
                if m[8] >= 0 {
                        year = q[m[8]:m[9]]
                }
                add(m[2], atoi(q[m[2]:m[3]]), month, year)
                if m[4] >= 0 {
                        add(m[4], atoi(q[m[4]:m[5]]), month, year)
                }
        }
        for _, m := range dotDateRe.FindAllStringSubmatchIndex(q, -1) {
                year := ""
                if m[6] >= 0 {
                        year = q[m[6]:m[7]]
                }
                add(m[2], atoi(q[m[2]:m[3]]), atoi(q[m[4]:m[5]]), year)
        }
        sort.SliceStable(all, func(i, j int) bool { return all[i].pos < all[j].pos })
        dates := make([]string, 0, len(all))
        for _, f := range all {
                dates = append(dates, f.date)
        }
        return firstTwo(dates)
}

// InvalidISODate возвращает дату вида YYYY-MM-DD из запроса, которой не бывает (например, 2026-02-30).
func InvalidISODate(q string) string {
        for _, m := range dateRe.FindAllString(q, -1) {
                if _, err := time.Parse("2006-01-02", m); err != nil {
                        return m
                }
        }
        return ""
}

func firstTwo(valid []string) (string, string) {
        switch len(valid) {
        case 0:
                return "", ""
        case 1:
                return valid[0], valid[0]
        default:
                if valid[1] < valid[0] {
                        return valid[1], valid[0]
                }
                return valid[0], valid[1]
        }
}

func atoi(s string) int {
        n, _ := strconv.Atoi(s)
        return n
}

func monthIndex(stem string) int {
        for i, s := range monthStems {
                if strings.HasPrefix(stem, s) {
                        return i + 1
                }
        }
        return 0
}

// makeDate собирает дату и отбрасывает несуществующие (30 февраля). Без года — ближайшая будущая.
func makeDate(day, month int, year string, today time.Time) (string, bool) {
        if month < 1 || month > 12 || day < 1 || day > 31 {
                return "", false
        }
        y := today.Year()
        explicit := year != ""
        if explicit {
                y = atoi(year)
                if y < 100 {
                        y += 2000
                }
        }
        d := time.Date(y, time.Month(month), day, 0, 0, 0, 0, time.UTC)
        if d.Day() != day || int(d.Month()) != month {
                return "", false
        }
        if !explicit && d.Format("2006-01-02") < today.Format("2006-01-02") {
                d = d.AddDate(1, 0, 0)
                if d.Day() != day { // 29 февраля в невисокосный год
                        return "", false
                }
        }
        return d.Format("2006-01-02"), true
}

// weekdays — основы названий дней недели (порядок фиксирован, чтобы результат был детерминированным).
var weekdays = []struct {
        stem string
        day  time.Weekday
}{
        {"понедельник", time.Monday},
        {"вторник", time.Tuesday},
        {"среду", time.Wednesday}, // не «сред»: иначе «среда разработки» станет датой
        {"четверг", time.Thursday},
        {"пятниц", time.Friday},
        {"суббот", time.Saturday},
        {"воскресень", time.Sunday},
}

// ParseWeekday — «в пятницу» → ближайшая будущая пятница от today (без учёта регистра).
// Понимает также «сегодня», «завтра» и «послезавтра».
func ParseWeekday(q string, today time.Time) (string, bool) {
        q = strings.ToLower(q)
        switch {
        case strings.Contains(q, "послезавтра"):
                return today.AddDate(0, 0, 2).Format("2006-01-02"), true
        case strings.Contains(q, "завтра"):
                return today.AddDate(0, 0, 1).Format("2006-01-02"), true
        case strings.Contains(q, "сегодня"):
                return today.Format("2006-01-02"), true
        }
        for _, w := range weekdays {
                if strings.Contains(q, w.stem) {
                        delta := (int(w.day) - int(today.Weekday()) + 7) % 7
                        if delta == 0 {
                                delta = 7
                        }
                        return today.AddDate(0, 0, delta).Format("2006-01-02"), true
                }
        }
        return "", false
}

        "ок": ackReply, "окей": ackReply, "ok": ackReply, "понятно": ackReply, "ясно": ackReply,

        "отмена": cancelReply, "отмени": cancelReply, "отменить": cancelReply, "отменяй": cancelReply,

// smallTalk возвращает ответ на приветствие, благодарность, «ок» или «отмени»; пусто — это не small talk.
// При смешанной реплике («спасибо, ок») приоритет у отмены, затем у благодарности и приветствия.
func smallTalk(q string) string {
        words := strings.FieldsFunc(strings.ToLower(q), func(r rune) bool { return !unicode.IsLetter(r) })
        if len(words) == 0 || len(words) > 4 {
                return ""
        }
        seen := map[string]bool{}
        for _, w := range words {
                reply, ok := smallTalkWords[w]
                if !ok {
                        return ""
                }
                seen[reply] = true
        }
        for _, reply := range []string{cancelReply, thanksReply, greetReply, ackReply} {
                if seen[reply] {
                        return reply
                }
        }
        return ""
}

// CreateAction создаёт черновик действия вручную (из UI). Подтверждение — отдельным запросом.
// Правила проверки общие с чатом и MCP: store.ValidateAction.
func (h *Handlers) CreateAction(w http.ResponseWriter, r *http.Request) {
        u, _ := auth.FromCtx(r.Context())
        var body actionReq
        if err := json.NewDecoder(r.Body).Decode(&body); err != nil {
                writeError(w, http.StatusBadRequest, "bad_request", "invalid json")
                return
        }
        if utf8.RuneCountInString(body.Reason) > store.MaxReasonRunes || utf8.RuneCountInString(body.Resource) > 50 {
                writeError(w, http.StatusBadRequest, "bad_request", "reason or resource too long")
                return
        }
        a := store.Absence{Type: body.Type, UserID: u.ID, TeamID: u.TeamID, Reason: body.Reason}
        switch body.Type {
        case store.ActionAbsence:
                a.DateFrom, a.DateTo = body.DateFrom, body.DateTo
        case store.ActionAccess:
                a.Resource = store.NormalizeResource(body.Resource)
        }
        if err := store.ValidateAction(a, time.Now()); err != nil {
                writeError(w, http.StatusBadRequest, "bad_request", err.Error())
                return
        }
        a.ID = h.Store.NextID(idPrefix(a.Type))
        created, err := h.Store.CreateDraft(a)
        if err != nil {
                writeError(w, http.StatusTooManyRequests, "too_many_drafts", "Слишком много неподтверждённых черновиков: подтвердите или отмените старые")
                return
        }
        writeJSON(w, http.StatusCreated, created)
}

// idPrefix — префикс номера: ABS-… для отгула, REQ-… для доступа.

                return "REQ"

        return "ABS"
}

// loadAction возвращает действие или пишет 404.

                writeError(w, http.StatusNotFound, "not_found", "action not found")
                return store.Absence{}, u, false
        }

}

// finish записывает переход в метрики, синхронизирует чек-лист и отвечает клиенту.
// Решение по заявке сохраняется на диск сразу (если включён STATE_FILE), а не по тику Persister.
func (h *Handlers) finish(w http.ResponseWriter, a store.Absence) {
        syncStep(h.Store, a)
        h.Store.RequestSave()
        h.Metrics.Inc("buddy_actions_total", "Действия пользователей по типу и статусу", "type", a.Type, "status", a.Status)
        writeJSON(w, http.StatusOK, a)
}

// ConfirmAction: draft → pending. Только владелец и только при наличии права роли.
func (h *Handlers) ConfirmAction(w http.ResponseWriter, r *http.Request) {
        a, u, ok := h.ownAction(w, r)
        if !ok {
                return
        }
        if !auth.Can(u, permissionFor(a.Type)) {
                writeError(w, http.StatusForbidden, "forbidden", "your role cannot confirm this action")
                return
        }
        // Повторная проверка: черновик мог устареть (дата прошла) или прийти в обход POST /actions.
        if a.Status == store.StatusDraft {
                if err := store.ValidateAction(a, time.Now()); err != nil {
                        writeError(w, http.StatusBadRequest, "bad_request", "draft is stale or invalid: "+err.Error())
                        return
                }
        }
        upd, ok := h.Store.TransitionBy(a.ID, []string{store.StatusDraft}, store.StatusPending, u.ID, "")
        if !ok {
                writeError(w, http.StatusConflict, "conflict", "already confirmed")
                return
        }
        h.finish(w, upd)
}

// CancelAction: владелец отменяет черновик или отправленную заявку.

        a, u, ok := h.ownAction(w, r)
        if !ok {
                return
        }
        upd, ok := h.Store.TransitionBy(a.ID, []string{store.StatusDraft, store.StatusPending}, store.StatusCancelled, u.ID, "")
        if !ok {
                writeError(w, http.StatusConflict, "conflict", "action is already decided")
                return
        }
        h.finish(w, upd)
}

// decide — решение руководителя по заявке своей команды: pending → approved | rejected.
func (h *Handlers) decide(w http.ResponseWriter, r *http.Request, status string) {
        a, u, ok := h.loadAction(w, r)
        if !ok {
                return
        }
        if !auth.Can(u, "action.approve") || a.TeamID == "" || a.TeamID != u.TeamID {
                writeError(w, http.StatusForbidden, "forbidden", "only the team lead can decide on this action")

        }
        if a.UserID == u.ID {
                writeError(w, http.StatusForbidden, "forbidden", "cannot decide on your own action")
                return
        }
        var body decisionReq
        if err := decodeOptional(r, &body); err != nil || utf8.RuneCountInString(body.Comment) > 500 {
                writeError(w, http.StatusBadRequest, "bad_request", "invalid json or comment too long")
                return
        }
        upd, ok := h.Store.TransitionBy(a.ID, []string{store.StatusPending}, status, u.ID, body.Comment)
        if !ok {
                writeError(w, http.StatusConflict, "conflict", "action is not pending")
                return
        }
        h.finish(w, upd)
}

func (h *Handlers) ApproveAction(w http.ResponseWriter, r *http.Request) {
        h.decide(w, r, store.StatusApproved)
}

func (h *Handlers) RejectAction(w http.ResponseWriter, r *http.Request) {
        h.decide(w, r, store.StatusRejected)
}

func Build(cfg config.Config, knowledgeDir string) (*App, error) {

        if err != nil {
                return nil, err
        }
        st := store.New()
        msgs := store.NewMessageStore()

        var persister *store.Persister
        if cfg.StateFile != "" {
                persister = &store.Persister{Path: cfg.StateFile, Store: st, Msgs: msgs}
                if err := persister.Load(); err != nil {
                        // Повреждённый снимок не должен мешать запуску: стартуем с чистого состояния.
                        slog.Error("state load failed, starting clean", "path", cfg.StateFile, "err", err)
                }
        }
        SeedDemo(st)

        reg := tools.NewRegistry()
        reg.Register(&tools.ChecklistGet{S: st})
        reg.Register(&tools.AbsenceCreateRequest{S: st})
        reg.Register(&tools.DirectoryFindOwner{})
        reg.Register(&tools.AbsenceGetPolicy{})
        reg.Register(&tools.AccessRequest{})

        m := metrics.New()
        m.Add("buddy_knowledge_chunks", "Фрагментов в базе знаний", float64(len(chunks)))

        ag := &agent.Agent{LLM: NewLLM(cfg), Index: rag.NewIndex(chunks), Tools: reg, Now: time.Now}
        if !cfg.LLMMock && cfg.LLMFallback {
                // Реальная модель недоступна → отвечает детерминированная резервная, ответ помечается degraded.
                ag.Fallback = llm.MockClient{}
                ag.OnFallback = func(err error) {
                        m.Inc("buddy_llm_fallback_total", "Переходы на резервную модель из-за сбоя основной")
                        slog.Warn("llm failed, using fallback", "err", err)
                }
        }

        chat := &ChatHandlers{Agent: ag, Store: st, Msgs: msgs, Broker: NewBroker(), Metrics: m}
        origins := DefaultOrigins
        if len(cfg.CORSOrigins) > 0 {
                origins = cfg.CORSOrigins
        }
        h := &Handlers{
                Agent: ag, Store: st, Chat: chat, Metrics: m,
                Limiter:      NewRateLimiter(cfg.RateLimitPerMin, time.Minute),
                WriteLimiter: NewRateLimiter(cfg.RateLimitPerMin, time.Minute),
        }
        return &App{Handler: NewRouterWithOrigins(h, origins), Persister: persister, Metrics: m, Store: st}, nil
}

// history собирает последние реплики диалога для модели (без текущего вопроса).
func (h *ChatHandlers) history(sessionID string) []llm.Message {
        var out []llm.Message
        for _, m := range h.Msgs.MessagesBySession(sessionID) {
                if m.Status != "done" || strings.TrimSpace(m.Content) == "" {
                        continue
                }
                out = append(out, llm.Message{Role: m.Role, Content: m.Content})
        }
        return out
}

// lastAssistant — последнее сообщение помощника в сессии.

        ms := h.Msgs.MessagesBySession(sessionID)
        for i := len(ms) - 1; i >= 0; i-- {

        return store.Message{}, false
}

// withDialogContext дополняет короткий ответ на уточнение помощника.
// Если помощник спросил дату отгула, а пользователь прислал только дату — продолжаем сценарий отгула.
func (h *ChatHandlers) withDialogContext(sessionID, q string) string {
        prev, ok := h.lastAssistant(sessionID)
        if !ok || prev.Result == "" {
                return q
        }
        var last agent.Response
        if json.Unmarshal([]byte(prev.Result), &last) != nil {
                return q
        }
        if last.Type != "draft" || last.Action != store.ActionAbsence || last.Args["action_id"] != "" {
                return q
        }
        if strings.Contains(strings.ToLower(q), "отгул") {
                return q
        }
        if from, _ := agent.ParseDatesAt(q, time.Now()); from != "" {
                return "Хочу отгул " + q
        }
        if _, ok := agent.ParseWeekday(q, time.Now()); ok {
                return "Хочу отгул " + q
        }
        return q
}

func (h *ChatHandlers) SendMessage(w http.ResponseWriter, r *http.Request) {
        sessionID, u, ok := h.ownSession(w, r)
        if !ok {
                return
        }
        var body sendMessageReq
        if err := json.NewDecoder(r.Body).Decode(&body); err != nil {
                writeError(w, http.StatusBadRequest, "bad_request", "invalid json")
                return
        }
        text := strings.TrimSpace(body.Q())
        if !validQuestion(w, text) {
                return
        }
        if h.Msgs.CountBySession(sessionID) >= maxMessagesPerSession {
                writeError(w, http.StatusTooManyRequests, "session_full", "Диалог слишком длинный — начните новый")
                return
        }
        if h.Msgs.PendingCount(sessionID) >= maxPendingPerSession {
                w.Header().Set("Retry-After", "3")
                writeError(w, http.StatusTooManyRequests, "rate_limited", "Помощник ещё отвечает на предыдущие вопросы, подождите")
                return
        }
        question := h.withDialogContext(sessionID, text)
        history := h.history(sessionID)

        userMsg := h.Msgs.CreateMessage(store.Message{
                ID: uuid.NewString(), SessionID: sessionID, Role: "user", Content: text,
                Status: "done", CreatedAt: time.Now(),
        })
        asstMsg := h.Msgs.CreateMessage(store.Message{
                ID: uuid.NewString(), SessionID: sessionID, Role: "assistant",
                Status: "pending", CreatedAt: time.Now(),
        })

        go h.runAgent(sessionID, asstMsg.ID, question, history, u)

        writeJSON(w, http.StatusAccepted, map[string]string{
                "user_message_id":      userMsg.ID,
                "assistant_message_id": asstMsg.ID,
                "messageId":            asstMsg.ID, // openapi
                "status":               "pending",
        })
}

type sendMessageReq struct {
        Question string `json:"question"`
        Text     string `json:"text"`
}

func (r sendMessageReq) Q() string {
        if r.Text != "" {
                return r.Text
        }
        return r.Question
}

func (h *ChatHandlers) runAgent(sessionID, msgID, question string, history []llm.Message, u auth.User) {
        defer func() {
                if rec := recover(); rec != nil {
                        slog.Error("agent panic", "session", sessionID, "message", msgID, "panic", rec)
                        h.Msgs.UpdateMessage(msgID, func(m *store.Message) { m.Status = "failed"; m.Content = "internal error" })
                        h.emit(sessionID, "message_done", msgID, map[string]any{"status": "failed"})
                }
        }()

        h.Msgs.UpdateMessage(msgID, func(m *store.Message) { m.Status = "in_progress" })
        h.emit(sessionID, "status", msgID, map[string]any{"status": "thinking"})

        ctx, cancel := context.WithTimeout(context.Background(), 60*time.Second)
        defer cancel()
        start := time.Now()
        resp, err := h.Agent.AskWithHistory(ctx, u, question, history)
        observeRun(h.Metrics, start)
        if err != nil {
                h.Metrics.Inc("buddy_agent_errors_total", "Сбои агента (LLM недоступна)")
                slog.Warn("agent failed", "session", sessionID, "message", msgID, "err", err)
                h.Msgs.UpdateMessage(msgID, func(m *store.Message) { m.Status = "failed"; m.Content = "agent failed" })
                h.emit(sessionID, "error", msgID, map[string]any{"error": "Помощник временно недоступен. Попробуйте ещё раз."})
                h.emit(sessionID, "message_done", msgID, map[string]any{"status": "failed"})
                return
        }
        saveDraft(h.Store, u, &resp)
        observe(h.Metrics, h.Store, u, question, resp)

        if resp.Type == "action" && resp.Result != nil {
                h.emit(sessionID, "tool_result", msgID, map[string]any{"ok": resp.Result.OK, "data": resp.Result})
        }

        b, _ := json.Marshal(resp)
        h.Msgs.UpdateMessage(msgID, func(m *store.Message) { m.Status = "done"; m.Result = string(b); m.Content = resp.Text })
        h.emit(sessionID, "message_done", msgID, map[string]any{"status": "done", "result": resp})
        slog.Info("agent done", "user", u.ID, "type", resp.Type, "action", resp.Action, "ms", time.Since(start).Milliseconds())
}

func writeSSE(w http.ResponseWriter, event, data string) {
        fmt.Fprintf(w, "event: %s\ndata: %s\n\n", event, data)
}

// Events — SSE-поток сессии. Подписка регистрируется ДО отправки заголовков,
// а готовые ответы, отправленные до подключения, отдаются повторно (фронт дедуплицирует по message_id).
func (h *ChatHandlers) Events(w http.ResponseWriter, r *http.Request) {
        sessionID, _, ok := h.ownSession(w, r)
        if !ok {
                return
        }
        fl, ok := w.(http.Flusher)
        if !ok {
                writeError(w, http.StatusInternalServerError, "sse_unsupported", "flusher not available")
                return
        }
        ch := h.Broker.Subscribe(sessionID)
        defer h.Broker.Unsubscribe(sessionID, ch)
        h.Metrics.Gauge("buddy_sse_streams", "Открытые SSE-потоки", 1)
        defer h.Metrics.Gauge("buddy_sse_streams", "Открытые SSE-потоки", -1)

        w.Header().Set("Content-Type", "text/event-stream")
        w.Header().Set("Cache-Control", "no-cache")
        w.Header().Set("X-Accel-Buffering", "no")
        w.WriteHeader(http.StatusOK)

        for _, m := range h.Msgs.MessagesBySession(sessionID) {
                if m.Role != "assistant" || (m.Status != "done" && m.Status != "failed") {
                        continue
                }
                ev := map[string]any{"type": "message_done", "message_id": m.ID, "messageId": m.ID, "status": m.Status}
                if m.Result != "" {
                        ev["result"] = json.RawMessage(m.Result)
                }
                b, _ := json.Marshal(ev)
                writeSSE(w, "message_done", string(b))
        }
        fl.Flush()

        tick := time.NewTicker(heartbeatEvery)
        defer tick.Stop()
        for {
                select {
                case <-r.Context().Done():
                        return
                case <-tick.C:
                        fmt.Fprint(w, ": ping\n\n")
                        fl.Flush()
                case ev, ok := <-ch:
                        if !ok {
                                return
                        }
                        writeSSE(w, ev.Event, ev.Data)
                        fl.Flush()
                }
        }
}

// saveDraft сохраняет черновик действия из ответа агента и кладёт его id в args.action_id.
// Черновик отгула без дат не сохраняется: агент сначала уточняет дату.
// Черновик проверяется теми же правилами, что и POST /actions (store.ValidateAction):
// некорректный не сохраняется, а пользователь получает объяснение (в MCP — isError).
func saveDraft(st *store.Store, u auth.User, resp *agent.Response) {
        saveDraftAt(st, u, resp, time.Now())
}

func saveDraftAt(st *store.Store, u auth.User, resp *agent.Response, now time.Time) {
        if resp.Type != "draft" {
                return
        }
        if resp.Args == nil {
                resp.Args = map[string]string{}
        }
        a := store.Absence{Type: resp.Action, UserID: u.ID, TeamID: u.TeamID, Reason: resp.Args["reason"]}
        switch resp.Action {
        case store.ActionAbsence:
                if resp.Args["date_from"] == "" {
                        return
                }
                a.DateFrom, a.DateTo = resp.Args["date_from"], resp.Args["date_to"]
                if a.DateTo == "" {
                        a.DateTo = a.DateFrom
                }
        case store.ActionAccess:
                a.Resource = store.NormalizeResource(resp.Args["resource"])
        default:
                return
        }
        buddy := auth.TeamContacts(u.TeamID).Buddy
        if err := store.ValidateAction(a, now); err != nil {
                rejectDraft(resp, draftProblem(a, u), buddy)
                return
        }
        a.ID = st.NextID(idPrefix(a.Type))
        created, err := st.CreateDraft(a)
        if err != nil {
                *resp = agent.Response{Type: "refusal", Buddy: buddy,
                        Text: "У вас уже много неподтверждённых черновиков. Подтвердите или отмените старые в разделе «Мои заявки» и повторите запрос."}
                return
        }
        resp.Args["action_id"] = created.ID
}

// rejectDraft оставляет пользователя без сохранённого черновика и объясняет почему.
// Отгул остаётся уточняющим черновиком без дат (диалог продолжается следующей датой),
// доступ превращается в отказ. В обоих случаях action_id нет — MCP вернёт isError.
func rejectDraft(resp *agent.Response, text, buddy string) {
        if resp.Action == store.ActionAbsence {
                resp.Args = map[string]string{"reason": "personal"}
                resp.Text = text
                return
        }
        *resp = agent.Response{Type: "refusal", Text: text, Buddy: buddy}
}

// draftProblem — человеческое объяснение, почему черновик не прошёл проверку.
// Наставнику и руководителю не советуем обратиться к наставнику (к себе или к подчинённому).
func draftProblem(a store.Absence, u auth.User) string {
        switch {
        case utf8.RuneCountInString(a.Reason) > store.MaxReasonRunes:
                return "Причина заявки слишком длинная — сократите её до 500 символов."
        case a.Type == store.ActionAccess && u.Role == auth.RoleIntern:
                return "Заявку можно оформить только на Jira, GitLab, Grafana или Git. По другим системам подскажет наставник " + auth.TeamContacts(u.TeamID).Buddy + "."
        case a.Type == store.ActionAccess:
                return "Заявку можно оформить только на Jira, GitLab, Grafana или Git. По другим системам обратитесь в IT Support."
        }
        return "Не могу оформить отгул на эти даты: дата не должна быть в прошлом, окончание — не раньше начала, а отгул — не длиннее 30 дней. Напишите другие даты."
}

// permissionFor — право, необходимое для подтверждения действия данного типа.
func permissionFor(actionType string) string {
        if actionType == store.ActionAccess {
                return "access.request"
        }
        return "absence.create"
}

// syncStep поддерживает связь «заявка на доступ ↔ шаг чек-листа»:
// отправлена → шаг ждёт (waiting); одобрена → выполнен; отклонена или отменена → снова todo.
func syncStep(st *store.Store, a store.Absence) {
        if a.Type != store.ActionAccess {
                return
        }
        switch a.Status {
        case store.StatusPending:
                st.LinkStep(a.UserID, "access", a.ID, store.StepWaiting)
        case store.StatusApproved:
                st.LinkStep(a.UserID, "access", a.ID, store.StepDone)
        case store.StatusRejected, store.StatusCancelled:
                st.UnlinkStep(a.UserID, a.ID)
        }
}

        Agent   *agent.Agent
        Store   *store.Store

        WriteLimiter *RateLimiter

type askReq struct {
        Question string `json:"question"`
        Text     string `json:"text"`
}

func (r askReq) Q() string {
        if r.Text != "" {
                return r.Text
        }
        return r.Question
}

// validQuestion проверяет вопрос и пишет 400, если он пустой или слишком длинный.
func validQuestion(w http.ResponseWriter, q string) bool {
        if q == "" {
                writeError(w, http.StatusBadRequest, "bad_request", "text required")
                return false
        }
        if utf8.RuneCountInString(q) > maxQuestionRunes {
                writeError(w, http.StatusBadRequest, "bad_request", "text too long (max 2000 chars)")
                return false
        }
        return true
}

func writeJSON(w http.ResponseWriter, status int, v any) {
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(v)
}

// decodeOptional читает необязательное JSON-тело: пустое тело — не ошибка.
func decodeOptional(r *http.Request, v any) error {
        if err := json.NewDecoder(r.Body).Decode(v); err != nil && !errors.Is(err, io.EOF) {
                return err
        }
        return nil
}

// observe учитывает ответ агента в метриках пользы и запоминает вопросы без ответа.
func observe(m *metrics.Registry, st *store.Store, u auth.User, question string, resp agent.Response) {
        m.Inc("buddy_agent_responses_total", "Ответы помощника по типу", "type", resp.Type)
        if resp.Degraded {
                m.Inc("buddy_agent_degraded_total", "Ответы, данные резервной моделью")
        }
        if resp.NoSource {
                st.RecordGap(u.TeamID, question)
                m.Inc("buddy_knowledge_gaps_total", "Вопросы, на которые в базе знаний нет ответа")
        }
}

// observeRun учитывает запуск агента и его длительность — одинаково для чата и синхронного /ask.
func observeRun(m *metrics.Registry, start time.Time) {
        m.Add("buddy_agent_duration_seconds_sum", "Суммарное время работы агента", time.Since(start).Seconds())
        m.Inc("buddy_agent_runs_total", "Запуски агента")
}

// Ask — синхронный вариант чата (удобен для проверки curl'ом).
func (h *Handlers) Ask(w http.ResponseWriter, r *http.Request) {
        u, ok := auth.FromCtx(r.Context())
        if !ok {
                writeError(w, http.StatusUnauthorized, "unauthorized", "no user")
                return
        }
        var body askReq
        if err := json.NewDecoder(r.Body).Decode(&body); err != nil {
                writeError(w, http.StatusBadRequest, "bad_request", "invalid json")
                return
        }
        if !validQuestion(w, body.Q()) {
                return
        }
        start := time.Now()
        resp, err := h.Agent.Ask(r.Context(), u, body.Q())
        observeRun(h.Metrics, start)
        if err != nil {
                h.Metrics.Inc("buddy_agent_errors_total", "Сбои агента (LLM недоступна)")
                writeError(w, http.StatusBadGateway, "agent_error", "Помощник временно недоступен, попробуйте ещё раз")
                return
        }
        saveDraft(h.Store, u, &resp)
        observe(h.Metrics, h.Store, u, body.Q(), resp)
        writeJSON(w, http.StatusOK, resp)
}

type progress struct {
        Items       []store.ChecklistItem `json:"items"`
        NextStep    string                `json:"next_step"`
        PercentDone int                   `json:"percent_done"`
}

func progressOf(items []store.ChecklistItem) progress {
        p := progress{Items: items}
        done := 0
        for _, it := range items {
                if it.Done {
                        done++
                } else if p.NextStep == "" && it.Status != store.StepWaiting {
                        p.NextStep = it.Title
                }
        }
        if len(items) > 0 {
                p.PercentDone = done * 100 / len(items)
        }
        return p
}

func (h *Handlers) OnboardingProgress(w http.ResponseWriter, r *http.Request) {
        u, _ := auth.FromCtx(r.Context())
        writeJSON(w, http.StatusOK, progressOf(h.Store.Checklist(u.ID)))
}

type stepReq struct {
        Status string `json:"status"`
}

func (h *Handlers) OnboardingStepUpdate(w http.ResponseWriter, r *http.Request) {
        u, _ := auth.FromCtx(r.Context())
        id := chi.URLParam(r, "id")
        // Тело необязательно; если передан status — поддерживаем только done.
        var body stepReq
        if err := decodeOptional(r, &body); err != nil {
                writeError(w, http.StatusBadRequest, "bad_request", "invalid json")
                return
        }
        if body.Status != "" && body.Status != store.StepDone {
                writeError(w, http.StatusBadRequest, "bad_request", "only status=done is supported")
                return
        }
        found, owned, locked := h.Store.MarkDoneOwned(u.ID, id)
        if !found {
                writeError(w, http.StatusNotFound, "not_found", "step not found")
                return
        }
        if !owned {
                writeError(w, http.StatusForbidden, "forbidden", "step belongs to another user")
                return
        }
        if locked {
                // Действие выполнено, когда его подтвердил руководитель: такой шаг закрывает только approve заявки.
                writeError(w, http.StatusConflict, "conflict", "Шаг закроется сам, когда руководитель одобрит заявку на доступ")
                return
        }
        h.Metrics.Inc("buddy_steps_done_total", "Шаги чек-листа, отмеченные выполненными")
        h.OnboardingProgress(w, r)
}

// MCP-сервер (Model Context Protocol, транспорт Streamable HTTP, JSON-RPC 2.0) поверх того же реестра
// инструментов, которым пользуется агент. Внешний агент или IDE подключается к POST /mcp и получает
// те же инструменты с теми же гарантиями:
//   - пользователь берётся из контекста запроса (X-Demo-User), а не из аргументов;
//   - право проверяется до вызова;
//   - мутирующие инструменты создают только ЧЕРНОВИК (isDraft=true, action_id), а не действие.

const mcpProtocolVersion = "2025-03-26"

type rpcRequest struct {
        JSONRPC string          `json:"jsonrpc"`
        ID      json.RawMessage `json:"id,omitempty"`
        Method  string          `json:"method"`
        Params  json.RawMessage `json:"params,omitempty"`
}

type rpcError struct {
        Code    int    `json:"code"`
        Message string `json:"message"`
}

type rpcResponse struct {
        JSONRPC string          `json:"jsonrpc"`
        ID      json.RawMessage `json:"id"`
        Result  any             `json:"result,omitempty"`
        Error   *rpcError       `json:"error,omitempty"`
}

const (
        rpcParseError     = -32700
        rpcInvalidRequest = -32600
        rpcMethodNotFound = -32601
        rpcInvalidParams  = -32602
)

func writeRPC(w http.ResponseWriter, id json.RawMessage, result any, rerr *rpcError) {
        if id == nil {
                id = json.RawMessage("null")
        }
        writeJSON(w, http.StatusOK, rpcResponse{JSONRPC: "2.0", ID: id, Result: result, Error: rerr})
}

type mcpContent struct {
        Type string `json:"type"`
        Text string `json:"text"`
}

type mcpToolResult struct {
        Content           []mcpContent `json:"content"`
        StructuredContent any          `json:"structuredContent,omitempty"`
        IsError           bool         `json:"isError"`
}

func toolText(structured any, isErr bool) mcpToolResult {
        b, _ := json.Marshal(structured)
        return mcpToolResult{Content: []mcpContent{{Type: "text", Text: string(b)}}, StructuredContent: structured, IsError: isErr}
}

func toolFail(msg string) mcpToolResult {
        return mcpToolResult{Content: []mcpContent{{Type: "text", Text: msg}}, IsError: true}
}

// mcpMethodLabel — значение метки method метрики: только из фиксированного набора, иначе "other".
// Имя метода приходит от клиента, и без этого каждый выдуманный метод заводил бы новый временной ряд.
func mcpMethodLabel(method string) string {
        switch method {
        case "initialize", "ping", "tools/list", "tools/call":
                return method
        }
        if strings.HasPrefix(method, "notifications/") {
                return "notifications"
        }
        return "other"
}

// MCP обрабатывает один JSON-RPC запрос.
func (h *Handlers) MCP(w http.ResponseWriter, r *http.Request) {
        u, _ := auth.FromCtx(r.Context())
        var req rpcRequest
        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
                writeRPC(w, nil, nil, &rpcError{rpcParseError, "parse error"})
                return
        }
        if req.JSONRPC != "2.0" || req.Method == "" {
                writeRPC(w, req.ID, nil, &rpcError{rpcInvalidRequest, "invalid request"})
                return
        }
        h.Metrics.Inc("buddy_mcp_requests_total", "Запросы к MCP-эндпоинту", "method", mcpMethodLabel(req.Method))
        // Уведомления (без id) ответа не требуют.
        if req.ID == nil {
                w.WriteHeader(http.StatusAccepted)
                return
        }

        switch req.Method {
        case "initialize":
                writeRPC(w, req.ID, map[string]any{
                        "protocolVersion": mcpProtocolVersion,
                        "capabilities":    map[string]any{"tools": map[string]any{"listChanged": false}},
                        "serverInfo":      map[string]string{"name": "ai-buddy-corporate-mocks", "version": "0.3.0"},
                        "instructions":    "Имитации корпоративных сервисов Авито для стажёра. Мутирующие инструменты создают черновик; подтверждение — POST /api/v1/actions/{id}/confirm.",
                }, nil)

        case "ping":
                writeRPC(w, req.ID, map[string]any{}, nil)

        case "tools/list":
                type mcpTool struct {
                        Name        string         `json:"name"`
                        Description string         `json:"description"`
                        InputSchema map[string]any `json:"inputSchema"`
                        Annotations map[string]any `json:"annotations"`
                }
                list := make([]mcpTool, 0)
                for _, s := range h.Agent.Tools.List() {
                        list = append(list, mcpTool{
                                Name: s.Name, Description: s.Description, InputSchema: s.InputSchema,
                                Annotations: map[string]any{
                                        "readOnlyHint":         !s.RequiresConfirmation,
                                        "requiresConfirmation": s.RequiresConfirmation,
                                        "permission":           s.Permission,
                                },
                        })
                }
                writeRPC(w, req.ID, map[string]any{"tools": list}, nil)

        case "tools/call":
                var p struct {
                        Name      string         `json:"name"`
                        Arguments map[string]any `json:"arguments"`
                }
                if err := json.Unmarshal(req.Params, &p); err != nil || p.Name == "" {
                        writeRPC(w, req.ID, nil, &rpcError{rpcInvalidParams, "params.name required"})
                        return
                }
                writeRPC(w, req.ID, h.mcpCall(r, u, p.Name, p.Arguments), nil)

        default:
                writeRPC(w, req.ID, nil, &rpcError{rpcMethodNotFound, "method not found: " + req.Method})
        }
}

func (h *Handlers) mcpCall(r *http.Request, u auth.User, name string, raw map[string]any) mcpToolResult {
        t, ok := h.Agent.Tools.Get(name)
        if !ok {
                return toolFail("unknown tool: " + name)
        }
        if !auth.Can(u, t.RequiredPermission()) {
                return toolFail("forbidden: role " + string(u.Role) + " lacks " + t.RequiredPermission())
        }
        args := make(map[string]string, len(raw))
        for k, v := range raw {
                if s, ok := v.(string); ok {
                        args[k] = s
                } else {
                        args[k] = fmt.Sprint(v)
                }
        }
        if t.RequiresConfirmation() {
                // Тот же путь, что у агента: черновик, который пользователь подтверждает сам.
                resp := h.Agent.DraftFor(u, "", name, args)
                if resp.Type != "draft" {
                        return toolFail(resp.Text)
                }
                saveDraft(h.Store, u, &resp)
                if resp.Args["action_id"] == "" {
                        return toolFail(resp.Text) // например, не указаны даты или дата в прошлом
                }
                return toolText(map[string]any{
                        "isDraft":   true,
                        "action_id": resp.Args["action_id"],
                        "action":    resp.Action,
                        "args":      resp.Args,
                        "next":      "POST /api/v1/actions/" + resp.Args["action_id"] + "/confirm",
                }, false)
        }
        res, err := tools.Run(r.Context(), t, u, args)
        if err != nil {
                return toolFail(err.Error())
        }
        return toolText(res, !res.OK)
}

// maxBodyBytes — лимит тела запроса: вопрос до 2000 символов с запасом.
const maxBodyBytes = 64 << 10

// DefaultDemoUser — учётка по умолчанию, если заголовок не передан (упрощённая авторизация прототипа).
const DefaultDemoUser = "intern.backend"

// DemoUser имитирует SSO: пользователь берётся из заголовка X-Demo-User. Query ?user= принимается
// только для GET-потока событий чата (EventSource не умеет передавать заголовки); на остальных маршрутах
// он игнорируется, иначе ссылка вида /actions/{id}/approve?user=lead.backend действовала бы от чужого имени.
// Неизвестный логин — 401.
func DemoUser(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                key := r.Header.Get("X-Demo-User")
                if key == "" && isEventStream(r) {
                        key = r.URL.Query().Get("user")
                }
                if key == "" {
                        key = DefaultDemoUser
                }
                u, ok := auth.DemoUsers[key]
                if !ok {
                        writeError(w, http.StatusUnauthorized, "unauthorized", "unknown demo user")
                        return
                }
                next.ServeHTTP(w, r.WithContext(auth.WithUser(r.Context(), u)))
        })
}

// isEventStream — GET /api/v1/chat/sessions/{id}/events: единственный маршрут, где допустим ?user=.
func isEventStream(r *http.Request) bool {
        if r.Method != http.MethodGet {
                return false
        }
        const prefix, suffix = "/api/v1/chat/sessions/", "/events"
        p := r.URL.Path
        if !strings.HasPrefix(p, prefix) || !strings.HasSuffix(p, suffix) {
                return false
        }
        id := p[len(prefix) : len(p)-len(suffix)]
        return id != "" && !strings.Contains(id, "/")
}

// SameOriginWrites — защита от CSRF: небезопасный запрос (POST, PATCH, PUT, DELETE) с заголовком Origin
// не из списка разрешённых CORS-origins отклоняется 403. Запросы без Origin (curl, тесты, MCP-клиенты)
// пропускаются: браузер всегда шлёт Origin на cross-origin POST, так что чужая страница сюда не попадёт.
func SameOriginWrites(origins []string) func(http.Handler) http.Handler {
        allowed := make(map[string]bool, len(origins))
        for _, o := range origins {
                allowed[strings.TrimRight(o, "/")] = true
        }
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        switch r.Method {
                        case http.MethodPost, http.MethodPatch, http.MethodPut, http.MethodDelete:
                                if o := r.Header.Get("Origin"); o != "" && !allowed[o] && !allowed["*"] {
                                        writeError(w, http.StatusForbidden, "forbidden", "cross-origin request blocked")
                                        return
                                }
                        }
                        next.ServeHTTP(w, r)
                })
        }
}

// LimitBody ограничивает размер тела запроса.
func LimitBody(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                if r.Body != nil {
                        r.Body = http.MaxBytesReader(w, r.Body, maxBodyBytes)
                }
                next.ServeHTTP(w, r)
        })
}

// SecurityHeaders — базовые защитные заголовки для API.
func SecurityHeaders(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                w.Header().Set("X-Content-Type-Options", "nosniff")
                w.Header().Set("Cache-Control", "no-store")
                next.ServeHTTP(w, r)
        })
}

// Observe пишет структурированный лог запроса (без тела — там могут быть вопросы стажёра),
// возвращает X-Request-Id и считает метрики по шаблону маршрута (а не по сырому пути).
func Observe(m *metrics.Registry) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        ww := middleware.NewWrapResponseWriter(w, r.ProtoMajor)
                        reqID := middleware.GetReqID(r.Context())
                        if reqID != "" {
                                ww.Header().Set("X-Request-Id", reqID)
                        }
                        start := time.Now()
                        next.ServeHTTP(ww, r)

                        route := "unmatched"
                        if rc := chi.RouteContext(r.Context()); rc != nil && rc.RoutePattern() != "" {
                                route = rc.RoutePattern()
                        }
                        status := ww.Status()
                        if status == 0 {
                                status = http.StatusOK
                        }
                        elapsed := time.Since(start)
                        m.Inc("buddy_http_requests_total", "HTTP-запросы", "method", r.Method, "route", route, "status", strconv.Itoa(status))
                        m.Add("buddy_http_request_duration_seconds_sum", "Суммарное время обработки HTTP-запросов", elapsed.Seconds(), "route", route)
                        slog.Info("http", "method", r.Method, "path", r.URL.Path, "status", status, "ms", elapsed.Milliseconds(), "req_id", reqID)
                })
        }
}

// RateLimiter — ограничение числа запросов пользователя в минуту (фиксированное окно).
// Защищает LLM-бюджет и сервер от «залипшей» кнопки и простого флуда.
type RateLimiter struct {
        mu     sync.Mutex
        limit  int
        window time.Duration
        now    func() time.Time
        hits   map[string]*bucket
}

type bucket struct {
        start time.Time
        n     int
}

// NewRateLimiter: limit запросов за window на пользователя. limit <= 0 — без ограничения (nil).
func NewRateLimiter(limit int, window time.Duration) *RateLimiter {
        if limit <= 0 {
                return nil
        }
        return &RateLimiter{limit: limit, window: window, now: time.Now, hits: map[string]*bucket{}}
}

// allow сообщает, можно ли выполнить запрос, и через сколько секунд повторить, если нельзя.
func (l *RateLimiter) allow(key string) (bool, int) {
        l.mu.Lock()
        defer l.mu.Unlock()
        now := l.now()
        b := l.hits[key]
        if b == nil || now.Sub(b.start) >= l.window {
                l.hits[key] = &bucket{start: now, n: 1}
                return true, 0
        }
        if b.n >= l.limit {
                retry := int(l.window.Seconds()-now.Sub(b.start).Seconds()) + 1
                return false, retry
        }
        b.n++
        return true, 0
}

// Middleware отвечает 429 с Retry-After, если пользователь превысил лимит. nil-лимитер ничего не ограничивает.
func (l *RateLimiter) Middleware(next http.Handler) http.Handler {
        if l == nil {
                return next
        }
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                u, _ := auth.FromCtx(r.Context())
                if ok, retry := l.allow(u.ID); !ok {
                        w.Header().Set("Retry-After", strconv.Itoa(retry))
                        writeError(w, http.StatusTooManyRequests, "rate_limited", "Слишком много запросов, попробуйте через "+strconv.Itoa(retry)+" с")
                        return
                }
                next.ServeHTTP(w, r)
        })
}

// DefaultOrigins — адреса фронта в dev-режиме (vite) и в docker.
var DefaultOrigins = []string{"http://localhost:5173", "http://127.0.0.1:5173", "http://localhost:3000"}

func NewRouter(h *Handlers) *chi.Mux {
        return NewRouterWithOrigins(h, DefaultOrigins)
}

func NewRouterWithOrigins(h *Handlers, origins []string) *chi.Mux {
        r := chi.NewRouter()
        r.Use(middleware.RequestID)
        r.Use(middleware.Recoverer)
        r.Use(Observe(h.Metrics))
        r.Use(SecurityHeaders)
        r.Use(cors.Handler(cors.Options{
                AllowedOrigins: origins,
                AllowedMethods: []string{"GET", "POST", "PATCH", "OPTIONS"},
                AllowedHeaders: []string{"Content-Type", "X-Demo-User", "Authorization", "Last-Event-ID"},
                ExposedHeaders: []string{"X-Request-Id", "Retry-After"},
                MaxAge:         300,
        }))
        // CSRF: CORS защищает только чтение ответа, а сам «простой» POST с чужой страницы браузер отправит.
        r.Use(SameOriginWrites(origins))
        r.Use(LimitBody)

        r.Get("/health", h.Health)
        if h.Metrics != nil {
                r.Get("/metrics", h.MetricsHandler)
        }

        r.Group(func(r chi.Router) {
                r.Use(DemoUser)

                r.Route("/api/v1", func(r chi.Router) {
                        r.Get("/me", h.Me)

                        r.Get("/onboarding/progress", h.OnboardingProgress)
                        r.Patch("/onboarding/steps/{id}", h.OnboardingStepUpdate)

                        // Действия: черновик → подтверждение → решение руководителя
                        r.Get("/actions", h.ListActions)
                        r.With(h.WriteLimiter.Middleware).Post("/actions", h.CreateAction)
                        r.Get("/actions/{id}", h.GetAction)
                        r.Post("/actions/{id}/confirm", h.ConfirmAction)
                        r.Post("/actions/{id}/cancel", h.CancelAction)
                        r.Post("/actions/{id}/approve", h.ApproveAction)
                        r.Post("/actions/{id}/reject", h.RejectAction)

type SSEEvent struct {
        Event string
        Data  string
}

type Broker struct {
        mu   sync.RWMutex
        subs map[string][]chan SSEEvent
}

func NewBroker() *Broker {
        return &Broker{subs: map[string][]chan SSEEvent{}}
}

func (b *Broker) Subscribe(sessionID string) chan SSEEvent {
        ch := make(chan SSEEvent, 16)
        b.mu.Lock()
        b.subs[sessionID] = append(b.subs[sessionID], ch)
        b.mu.Unlock()
        return ch
}

func (b *Broker) Unsubscribe(sessionID string, ch chan SSEEvent) {
        b.mu.Lock()
        defer b.mu.Unlock()
        list := b.subs[sessionID]
        for i, c := range list {
                if c == ch {
                        b.subs[sessionID] = append(list[:i], list[i+1:]...)
                        close(ch)
                        return
                }
        }
}

func (b *Broker) Publish(sessionID string, ev SSEEvent) {
        b.mu.RLock()
        defer b.mu.RUnlock()
        for _, ch := range b.subs[sessionID] {
                select {
                case ch <- ev:
                default:
                }
        }
}

        u, ok := h.teamUser(w, r)
        if !ok {
                return

        s := teamSummary{Actions: map[string]int{}}
        total := 0
        for _, intern := range auth.TeamInterns(u.TeamID) {
                p := progressOf(h.Store.Checklist(intern.ID))
                s.Interns++
                total += p.PercentDone
                for _, it := range p.Items {
                        if it.Status == store.StepWaiting {
                                s.StepsWaiting++
                        }
                }
        }
        if s.Interns > 0 {
                s.AvgPercentDone = total / s.Interns
        }

                s.Actions[a.Status]++
        }
        for _, g := range h.Store.Gaps(u.TeamID) {
                s.UnansweredTopics++
                s.UnansweredRequests += g.Count
        }
        writeJSON(w, http.StatusOK, s)

func LoadDotEnv(path string) {
        f, err := os.Open(path)
        if err != nil {
                return
        }
        defer f.Close()
        sc := bufio.NewScanner(f)
        for sc.Scan() {
                line := strings.TrimSpace(sc.Text())
                if line == "" || strings.HasPrefix(line, "#") {
                        continue
                }
                parts := strings.SplitN(line, "=", 2)
                if len(parts) != 2 {
                        continue
                }
                k := strings.TrimSpace(parts[0])
                v := strings.TrimSpace(parts[1])
                if os.Getenv(k) == "" {
                        _ = os.Setenv(k, v)
                }
        }
}

// MockClient — детерминированная «LLM» для демо и тестов: работает без ключа и интернета.
// Смотрит только на строку QUESTION (не на CONTEXT), а ответ по базе берёт из первого фрагмента CONTEXT.
type MockClient struct{}

func extractQuestion(prompt string) string {
        const marker = "QUESTION:"
        i := strings.Index(prompt, marker)
        if i < 0 {
                return strings.ToLower(prompt)
        }
        rest := prompt[i+len(marker):]
        if j := strings.Index(rest, "\n"); j >= 0 {
                rest = rest[:j]
        }
        return strings.ToLower(strings.TrimSpace(rest))
}

// firstChunk возвращает текст первого фрагмента из блока CONTEXT без префикса [id].
func firstChunk(prompt string) string {
        const marker = "CONTEXT:\n"
        i := strings.Index(prompt, marker)
        if i < 0 {
                return ""
        }
        rest := prompt[i+len(marker):]
        if j := strings.Index(rest, "\n["); j >= 0 {
                rest = rest[:j]
        }
        if strings.HasPrefix(rest, "[") {
                if k := strings.Index(rest, "] "); k >= 0 {
                        rest = rest[k+2:]
                }
        }
        return strings.TrimSpace(rest)
}

func containsAny(s string, subs ...string) bool {
        for _, x := range subs {
                if strings.Contains(s, x) {
                        return true
                }
        }
        return false
}

func toJSON(v any) (string, error) {
        b, err := json.Marshal(v)
        return string(b), err
}

func (MockClient) Complete(_ context.Context, msgs []Message) (string, error) {
        prompt := msgs[len(msgs)-1].Content
        q := extractQuestion(prompt)

        refuse := func(text string) (string, error) {
                return toJSON(map[string]any{"type": "refusal", "text": text})
        }
        toolCall := func(tool string, args map[string]string) (string, error) {
                return toJSON(map[string]any{"type": "tool_call", "tool": tool, "args": args})
        }

        if containsAny(q, "ignore previous", "игнорируй", "покажи все", "все отгулы", "чужие",
                "забудь все", "забудь инструкц", "системный промпт", "system prompt", "ты теперь", "system:") {
                // Имя наставника мок не знает (оно зависит от команды) — совет обратиться к нему добавляет агент.
                return refuse("Не могу выполнить это действие.")
        }

        // Слова сравниваем по началу слова: « дай» ловит «дай» и «дайте», но не «выдают».
        w := words(q)
        asks := hasWord(w, "где", "кто", "сколько", "когда", "зачем", "почему", "что такое", "как ", "какой", "какая", "какие")

        if containsAny(q, "отгул", "отпуск", "выходной") {
                // «Как отменить заявку на отгул?» — вопрос по базе знаний, а не новый черновик.
                if hasWord(w, "отмен") {
                        return answerFromContext(prompt)
                }
                intent := hasWord(w, "хочу", "возьму", "взять", "оформи ", "оформите", "нужен", "нужна", "запиши", "поставь")
                info := asks || hasWord(w, "порядок", "правил", "оформить", "оформлени", "положен")
                if info && !intent {
                        return toolCall("absence_get_policy", map[string]string{})
                }
                return toolCall("absence_create_request", map[string]string{"reason": "personal"})
        }

        if containsAny(q, "доступ", "jira", "gitlab", "grafana", "джир", "гитлаб", "графан") {
                // Черновик — только при явной просьбе. Справочный вопрос («Где смотреть логи в Grafana?»,
                // «Кто выдаёт доступы?») отвечаем по CONTEXT, иначе каждый вопрос плодит заявки.
                imperative := hasWord(w, "запроси", "дай", "выдай", "оформи ", "оформите", "открой", "нужен", "нужна", "хочу", "сделай")
                direct := !asks && hasWord(w, "доступ к", "доступ в", "доступа к", "доступа в")
                if imperative || direct {
                        return toolCall("access_request", map[string]string{"resource": accessResource(q, w), "reason": "onboarding"})
                }
                return answerFromContext(prompt)
        }

        if hasWord(w, "чек-лист", "чеклист", "на этой неделе", "на следующей неделе", "что делать", "что мне делать",
                "следующий шаг", "что мне сделать", "прогресс") {
                return toolCall("checklist_get", map[string]string{})
        }

        return answerFromContext(prompt)
}

// answerFromContext отвечает первым фрагментом CONTEXT или отказывает, если контекста нет.
func answerFromContext(prompt string) (string, error) {
        if txt := firstChunk(prompt); txt != "" {
                return toJSON(map[string]any{"type": "answer", "text": txt})
        }
        return toJSON(map[string]any{"type": "refusal", "text": "Не нашёл в доступных источниках."})
}

// accessResource — система из просьбы о доступе. Неизвестную систему («Confluence») возвращаем как есть,
// а не подменяем на Jira: агент честно ответит, что на неё заявку не оформить. Пусто — система не названа.
func accessResource(q, w string) string {
        switch {
        case containsAny(q, "gitlab", "гитлаб"):
                return "gitlab"
        case containsAny(q, "grafana", "графан"):
                return "grafana"
        case containsAny(q, "jira", "джир"):
                return "jira"
        case hasWord(w, "git ", "гит "):
                return "git"
        }
        for _, p := range []string{" доступ к ", " доступ в ", " доступа к ", " доступа в ", " доступ до "} {
                if i := strings.Index(w, p); i >= 0 {
                        if f := strings.Fields(w[i+len(p):]); len(f) > 0 {
                                return f[0]
                        }
                }
        }
        return ""
}

// words нормализует вопрос в « слово слово … » (знаки препинания → пробелы, дефис сохраняется).
func words(q string) string {
        f := strings.FieldsFunc(q, func(r rune) bool {
                return !unicode.IsLetter(r) && !unicode.IsDigit(r) && r != '-'
        })
        return " " + strings.Join(f, " ") + " "
}

// hasWord — встречается ли фраза с начала слова («нужн» найдёт «нужно»; «как » — только «как»).
func hasWord(w string, phrases ...string) bool {
        for _, p := range phrases {
                if strings.Contains(w, " "+p) {
                        return true
                }
        }
        return false
}

// OpenAIClient — клиент OpenAI-совместимого /chat/completions с повторами при временных сбоях.
type OpenAIClient struct {
        BaseURL, APIKey, Model string
        HTTP                   *http.Client
        MaxRetries             int           // повторы после первой попытки; по умолчанию 2
        Backoff                time.Duration // пауза перед первым повтором, дальше ×3; по умолчанию 300 мс
}

type respFormat struct {
        Type string `json:"type"`
}

type req struct {
        Model          string     `json:"model"`
        Messages       []Message  `json:"messages"`
        Temperature    float64    `json:"temperature"`
        ResponseFormat respFormat `json:"response_format"`
}

type resp struct {
        Choices []struct {
                Message Message `json:"message"`
        } `json:"choices"`
}

// retryable — временная ошибка: имеет смысл повторить.
type retryable struct{ err error }

func (r retryable) Error() string { return r.err.Error() }

func (c *OpenAIClient) Complete(ctx context.Context, msgs []Message) (string, error) {
        body, _ := json.Marshal(req{
                Model: c.Model, Messages: msgs, Temperature: 0.2,
                ResponseFormat: respFormat{Type: "json_object"},
        })
        retries := c.MaxRetries
        if retries == 0 {
                retries = 2
        }
        backoff := c.Backoff
        if backoff == 0 {
                backoff = 300 * time.Millisecond
        }

        var lastErr error
        for attempt := 0; attempt <= retries; attempt++ {
                if attempt > 0 {
                        select {
                        case <-ctx.Done():
                                return "", ctx.Err()
                        case <-time.After(backoff):
                        }
                        backoff *= 3
                }
                out, err := c.once(ctx, body)
                if err == nil {
                        return out, nil
                }
                lastErr = err
                if _, ok := err.(retryable); !ok || ctx.Err() != nil {
                        break
                }
        }
        return "", lastErr
}

func (c *OpenAIClient) once(ctx context.Context, body []byte) (string, error) {
        r, err := http.NewRequestWithContext(ctx, "POST", c.BaseURL+"/chat/completions", bytes.NewReader(body))
        if err != nil {
                return "", err
        }
        r.Header.Set("Content-Type", "application/json")
        r.Header.Set("Authorization", "Bearer "+c.APIKey)
        res, err := c.HTTP.Do(r)
        if err != nil {
                return "", retryable{err} // сеть, таймаут
        }
        defer res.Body.Close()
        if res.StatusCode == http.StatusTooManyRequests || res.StatusCode >= 500 {
                return "", retryable{fmt.Errorf("llm status %d", res.StatusCode)}
        }
        if res.StatusCode >= 300 {
                return "", fmt.Errorf("llm status %d", res.StatusCode) // 4xx — повтор не поможет
        }
        var out resp
        if err := json.NewDecoder(io.LimitReader(res.Body, 1<<20)).Decode(&out); err != nil {
                return "", err
        }
        if len(out.Choices) == 0 {
                return "", fmt.Errorf("empty choices")
        }
        return out.Choices[0].Message.Content, nil
}

// Package metrics — минимальный реестр метрик в формате Prometheus без внешних зависимостей.
// Покрывает технические метрики (HTTP, LLM) и метрики пользы (ответы, отказы, действия).
package metrics

import (
        "fmt"
        "io"
        "sort"
        "strings"
        "sync"
)

type series struct {
        help   string
        kind   string // counter | gauge
        values map[string]float64
}

type Registry struct {
        mu     sync.Mutex
        series map[string]*series
}

func New() *Registry { return &Registry{series: map[string]*series{}} }

func labelKey(labels []string) string {
        if len(labels) == 0 {
                return ""
        }
        var b strings.Builder
        for i := 0; i+1 < len(labels); i += 2 {
                if i > 0 {
                        b.WriteByte(',')
                }
                fmt.Fprintf(&b, "%s=%q", labels[i], labels[i+1])
        }
        return b.String()
}

func (r *Registry) add(name, kind, help string, v float64, labels []string) {
        if r == nil {
                return
        }
        r.mu.Lock()
        defer r.mu.Unlock()
        s := r.series[name]
        if s == nil {
                s = &series{help: help, kind: kind, values: map[string]float64{}}
                r.series[name] = s
        }
        s.values[labelKey(labels)] += v
}

// Inc увеличивает счётчик на 1. labels — пары ключ, значение.
func (r *Registry) Inc(name, help string, labels ...string) { r.add(name, "counter", help, 1, labels) }

// Add увеличивает счётчик на v (например, суммарные секунды).
func (r *Registry) Add(name, help string, v float64, labels ...string) {
        r.add(name, "counter", help, v, labels)
}

// Gauge изменяет текущее значение на delta (например, число открытых SSE-потоков).
func (r *Registry) Gauge(name, help string, delta float64, labels ...string) {
        r.add(name, "gauge", help, delta, labels)
}

// Value возвращает текущее значение серии (для тестов и сводки).
func (r *Registry) Value(name string, labels ...string) float64 {
        if r == nil {
                return 0
        }
        r.mu.Lock()
        defer r.mu.Unlock()
        if s := r.series[name]; s != nil {
                return s.values[labelKey(labels)]
        }
        return 0
}

// Sum — сумма по всем наборам меток серии.
func (r *Registry) Sum(name string) float64 {
        if r == nil {
                return 0
        }
        r.mu.Lock()
        defer r.mu.Unlock()
        total := 0.0
        if s := r.series[name]; s != nil {
                for _, v := range s.values {
                        total += v
                }
        }
        return total
}

// Write выводит метрики в текстовом формате Prometheus (детерминированный порядок).
func (r *Registry) Write(w io.Writer) {
        r.mu.Lock()
        defer r.mu.Unlock()
        names := make([]string, 0, len(r.series))
        for n := range r.series {
                names = append(names, n)
        }
        sort.Strings(names)
        for _, n := range names {
                s := r.series[n]
                fmt.Fprintf(w, "# HELP %s %s\n# TYPE %s %s\n", n, s.help, n, s.kind)
                keys := make([]string, 0, len(s.values))
                for k := range s.values {
                        keys = append(keys, k)
                }
                sort.Strings(keys)
                for _, k := range keys {
                        if k == "" {
                                fmt.Fprintf(w, "%s %g\n", n, s.values[k])
                        } else {
                                fmt.Fprintf(w, "%s{%s} %g\n", n, k, s.values[k])
                        }
                }
        }
}

type Chunk struct {
        ID         string
        Source     string
        Stage      string
        Title      string // заголовок раздела: индексируется вместе с текстом
        TeamID     string
        Audience   []string
        Text       string
        Disclaimer string
}

var headerRe = regexp.MustCompile(`^#+\s+(.*)$`)

func ChunkMarkdown(source, content string) []Chunk {
        lines := strings.Split(content, "\n")
        var chunks []Chunk
        var cur Chunk
        var buf []string

        flush := func() {
                text := strings.TrimSpace(strings.Join(buf, "\n"))
                if text == "" {
                        return
                }
                cur.Text = text
                cur.Source = source
                cur.ID = source + "#" + cur.Stage + "#" + itoa(len(chunks))
                chunks = append(chunks, cur)
                buf = buf[:0]
        }

        for _, ln := range lines {
                if m := headerRe.FindStringSubmatch(ln); m != nil {
                        flush()
                        cur = Chunk{Stage: m[1], Title: m[1]}
                        continue
                }
                if strings.HasPrefix(strings.TrimSpace(ln), "<!--") {
                        // Метаданные раздела; прочие комментарии (например, дисклеймер файла) не индексируются.
                        if strings.Contains(ln, "stage:") {
                                cur.Stage = extract(ln, "stage")
                                cur.TeamID = extract(ln, "team")
                                cur.Audience = splitCSV(extract(ln, "audience"))
                        }
                        continue
                }
                buf = append(buf, ln)
        }
        flush()
        return chunks
}

func extract(s, key string) string {
        re := regexp.MustCompile(key + `:\s*([^\s]+)`)
        if m := re.FindStringSubmatch(s); len(m) == 2 {
                return m[1]
        }
        return ""
}

func splitCSV(s string) []string {
        if s == "" {
                return nil
        }
        parts := strings.Split(s, ",")
        out := make([]string, 0, len(parts))
        for _, p := range parts {
                if v := strings.TrimSpace(p); v != "" {
                        out = append(out, v)
                }
        }
        return out
}

func itoa(i int) string {
        if i == 0 {
                return "0"
        }
        var b []byte
        for i > 0 {
                b = append([]byte{byte('0' + i%10)}, b...)
                i /= 10
        }
        return string(b)
}

func tokenize(s string) []string {
        s = strings.ToLower(s)
        return strings.FieldsFunc(s, func(r rune) bool {
                if r == 'ё' {
                        return false
                }
                return !(r >= 'a' && r <= 'z') && !(r >= 'а' && r <= 'я') && !(r >= '0' && r <= '9')
        })
}

// stopwords — служебные и вопросительные слова, которые не несут смысла для поиска.
// Без них «Сколько платят в Авито?» находило бы фрагменты по предлогу «в».
var stopwords = map[string]bool{}

                такое такой такая такие смотреть посмотреть

        }

// synonyms — транслитерация, опечатки и разговорные формы → слово из базы знаний (по префиксу).
// Применяется и к запросу, и к фрагментам, поэтому обе стороны приходят к одной форме.
var synonyms = []struct{ prefix, canon string }{
        {"стэндап", "стендап"},
        {"фронтенд", "frontend"}, {"фронт", "frontend"},
        {"бэкенд", "backend"}, {"бекенд", "backend"},
        {"мердж", "merge"}, {"мерж", "merge"},
        {"гитлаб", "gitlab"}, {"джир", "jira"}, {"графан", "grafana"},
        {"поставить", "установить"}, {"ставить", "установить"},
        {"поднима", "поднять"}, {"подня", "поднять"},
}

func canonical(t string) string {
        for _, s := range synonyms {
                if strings.HasPrefix(t, s.prefix) {
                        return s.canon
                }
        }
        return t
}

// terms — токены без стоп-слов и без слишком коротких (1 символ) слов, с приведением синонимов.
func terms(s string) []string {
        toks := tokenize(s)
        out := toks[:0]
        for _, t := range toks {
                if len([]rune(t)) < 2 || stopwords[t] || isNumber(t) {
                        continue
                }
                out = append(out, canonical(t))
        }
        return out
}

// Meaningful — есть ли в тексте значимые слова (не только стоп-слова, числа и знаки).
func Meaningful(s string) bool { return len(terms(s)) > 0 }

// endings — типовые окончания русских слов (облегчённый стемминг), от длинных к коротким.
var endings = []string{
        "ться", "тся", "ает", "яет", "ают", "яют", "аёт", "ить", "ать", "ять", "еть",
        "ами", "ями", "ого", "его", "ему", "ому", "ыми", "ими",
        "ую", "юю", "ая", "яя", "ой", "ей", "ий", "ый", "ом", "ем", "ов", "ев", "ах", "ях", "ам", "ям", "ия", "ие", "ью",
        "ёт", "ет", "ут", "ют", "ит", "ат", "ят", "ть",
        "у", "а", "я", "ы", "и", "е", "о", "ь", "ю",
}

// stem отрезает окончание, оставляя основу не короче трёх букв: «базу», «база» → «баз».
func stem(w string) string {
        r := []rune(w)
        for _, e := range endings {
                re := []rune(e)
                if len(r)-len(re) >= 3 && strings.HasSuffix(w, e) {
                        return string(r[:len(r)-len(re)])
                }
        }
        return w
}

func isNumber(s string) bool {
        for _, r := range s {
                if r < '0' || r > '9' {
                        return false
                }
        }
        return true
}

func NewIndex(chunks []Chunk) *Index {
        idx := &Index{chunks: chunks, df: map[string]int{}}
        for _, c := range chunks {
                tf := map[string]int{}
                toks := terms(c.Title + " " + c.Text + " " + c.Stage)
                for _, t := range toks {
                        tf[t]++
                }
                for t := range tf {
                        idx.df[t]++
                }
                idx.tf = append(idx.tf, tf)
                idx.lens = append(idx.lens, len(toks))
        }
        return idx
}

type Hit struct {
        Chunk Chunk
        Score float64
}

func prefixMatch(a, b string) bool {
        if a == b {
                return true
        }
        ra, rb := []rune(a), []rune(b)
        n := 5
        if len(ra) < n {
                n = len(ra)
        }
        if len(rb) < n {
                n = len(rb)
        }
        if n < 4 {
                return false
        }
        return string(ra[:n]) == string(rb[:n])
}

// termMatch — слово запроса и слово фрагмента совпадают: точно, по общему префиксу
// или по основе после отрезания окончания («базу» ~ «база»).
func termMatch(a, b string) bool {
        if prefixMatch(a, b) {
                return true
        }
        sa, sb := stem(a), stem(b)
        return len([]rune(sa)) >= 3 && sa == sb
}

const (
        minScore = 0.5
        // minCoverage — доля «веса» запроса (сумма IDF значимых слов), которая должна найтись во фрагменте.
        // Частые слова («стажёр», «заявка», «получить») весят мало и не тянут нерелевантный ответ.
        minCoverage = 0.6
        // shortQuery — в запросе из стольких значимых слов (и меньше) должны совпасть все.
        shortQuery = 2
)

func idfOf(n, df float64) float64 {
        return math.Log(1 + (n-df+0.5)/(df+0.5))
}

// allowed — deny-by-default: если у чанка пустой audience — невидим.
func allowed(c Chunk, role string) bool {
        if len(c.Audience) == 0 {
                return false
        }
        for _, a := range c.Audience {
                if a == role || a == "all" {
                        return true
                }
        }
        return false
}

func (idx *Index) Search(q string, teamID string, role string, k int) []Hit {
        return idx.search(q, teamID, func(c Chunk) bool { return allowed(c, role) }, k)
}

// Restricted сообщает, что релевантный фрагмент своей команды есть, но он адресован другой аудитории
// (например, «только для наставников»). Это не пробел базы знаний: ответ есть, просто роль не та.
// Сам фрагмент наружу не отдаётся — только факт его существования.
func (idx *Index) Restricted(q string, teamID string, role string) bool {
        hits := idx.search(q, teamID, func(c Chunk) bool { return len(c.Audience) > 0 && !allowed(c, role) }, 1)
        return len(hits) > 0
}

func (idx *Index) search(q string, teamID string, visible func(Chunk) bool, k int) []Hit {
        qt := terms(q)
        n := float64(len(idx.chunks))
        avg := 0.0
        for _, l := range idx.lens {
                avg += float64(l)
        }
        if n > 0 {
                avg /= n
        }
        if avg == 0 {
                avg = 1
        }

        if len(qt) == 0 {
                return nil
        }
        // Вес слова запроса — его IDF по базе: в скольких фрагментах оно встречается.
        // Слово, которого нет в базе, весит как редкое (df=1): оно важно, и без него ответа нет.
        weights := make([]float64, len(qt))
        total := 0.0
        for j, t := range qt {
                df := 0
                for i := range idx.chunks {
                        for ct := range idx.tf[i] {
                                if termMatch(t, ct) {
                                        df++
                                        break
                                }
                        }
                }
                if df == 0 {
                        df = 1
                }
                weights[j] = idfOf(n, float64(df))
                total += weights[j]
        }

        var hits []Hit
        for i, c := range idx.chunks {
                if teamID != "" && c.TeamID != "" && c.TeamID != teamID {
                        continue
                }
                if !visible(c) {
                        continue
                }
                score, covered := 0.0, 0.0
                matched := 0
                for j, t := range qt {
                        // Все словоформы слова запроса во фрагменте складываются в одну частоту,
                        // а вес берётся у слова запроса: редкая словоформа («наставником») не перевешивает тему фрагмента.
                        f := 0
                        for ct, cnt := range idx.tf[i] {
                                if termMatch(t, ct) {
                                        f += cnt
                                }
                        }
                        if f == 0 {
                                continue
                        }
                        matched++
                        covered += weights[j]
                        score += weights[j] * (float64(f) * 2.2) / (float64(f) + 1.2*(0.25+0.75*float64(idx.lens[i])/avg))
                }
                if score < minScore {
                        continue
                }
                if len(qt) <= shortQuery && matched < len(qt) {
                        continue
                }
                if total > 0 && covered/total < minCoverage {
                        continue
                }
                hits = append(hits, Hit{Chunk: c, Score: score})
        }
        sort.Slice(hits, func(i, j int) bool { return hits[i].Score > hits[j].Score })
        if len(hits) > k {
                hits = hits[:k]
        }
        return hits
}

func LoadDir(dir string) ([]Chunk, error) {

        if err != nil {

        }

                if e.IsDir() || filepath.Ext(e.Name()) != ".md" {

// Gap — вопрос, на который в базе знаний не нашлось ответа.
// Копится без привязки к пользователю: наставник видит, что дописать в базу, но не кто спрашивал.
type Gap struct {
        Question string    `json:"question"`
        Count    int       `json:"count"`
        LastAt   time.Time `json:"last_at"`
}

const (
        maxGapsPerTeam = 200
        maxGapLen      = 200
)

func normalizeQuestion(q string) string {
        q = strings.ToLower(strings.Join(strings.Fields(q), " "))
        return strings.TrimRight(q, "?!. ")
}

var (
        emailRe = regexp.MustCompile(`[\p{L}\d._%+\-]+@[\p{L}\d.\-]+\.\p{L}{2,}`)
        // numberRe — число с разделителями: телефон «+7 (999) 123-45-67», паспорт «4510 123456», счёт.
        numberRe = regexp.MustCompile(`[+(]?\d[\d\s()\-]*\d\)?`)
)

// minMaskedDigits — последовательность из стольких цифр и больше считается персональными данными.
const minMaskedDigits = 6

// MaskPII убирает из текста персональные данные: email → [email], телефоны, номера паспортов и прочие
// последовательности из 6+ цифр (с пробелами, дефисами, скобками, плюсом) → [число].
// Пробелы базы знаний видит наставник — личные данные стажёра туда попадать не должны.
func MaskPII(q string) string {
        q = emailRe.ReplaceAllString(q, "[email]")
        return numberRe.ReplaceAllStringFunc(q, func(m string) string {
                n := 0
                for _, r := range m {
                        if unicode.IsDigit(r) {
                                n++
                        }
                }
                if n < minMaskedDigits {
                        return m
                }
                return "[число]"
        })
}

// RecordGap запоминает вопрос без ответа для команды (с замаскированными персональными данными).
func (s *Store) RecordGap(teamID, question string) {
        question = MaskPII(question)
        key := normalizeQuestion(question)
        if key == "" {
                return
        }
        if r := []rune(question); len(r) > maxGapLen {
                question = string(r[:maxGapLen]) + "…"
        }
        s.mu.Lock()
        defer s.mu.Unlock()
        team := s.gaps[teamID]
        if team == nil {
                team = map[string]*Gap{}
                s.gaps[teamID] = team
        }
        g := team[key]
        if g == nil {
                if len(team) >= maxGapsPerTeam {
                        return // защита от разрастания
                }
                g = &Gap{Question: strings.TrimSpace(question)}
                team[key] = g
        }
        g.Count++
        g.LastAt = s.now()
        s.touch()
}

// Gaps — вопросы без ответа, самые частые сверху.
func (s *Store) Gaps(teamID string) []Gap {
        s.mu.RLock()
        out := make([]Gap, 0, len(s.gaps[teamID]))
        for _, g := range s.gaps[teamID] {
                out = append(out, *g)
        }
        s.mu.RUnlock()
        sort.Slice(out, func(i, j int) bool {
                if out[i].Count != out[j].Count {
                        return out[i].Count > out[j].Count
                }
                return out[i].Question < out[j].Question
        })
        return out
}

        defer s.mu.Unlock()
        sess := Session{ID: id, UserID: userID, CreatedAt: time.Now()}
        s.sessions[id] = sess
        s.version.Add(1)
        return sess

        s.mu.RLock()
        defer s.mu.RUnlock()

func (s *MessageStore) CreateMessage(m Message) Message {
        s.mu.Lock()
        defer s.mu.Unlock()
        s.messages[m.ID] = m
        s.bySess[m.SessionID] = append(s.bySess[m.SessionID], m.ID)
        s.version.Add(1)
        return m
}

        s.mu.Lock()

                return Message{}, false
        }
        fn(&m)
        s.messages[id] = m
        s.version.Add(1)
        return m, true

        s.mu.RLock()

// snapshot — состояние приложения на диске. Формат версионируется полем V.
type snapshot struct {
        V         int                        `json:"v"`
        SavedAt   time.Time                  `json:"saved_at"`
        Actions   map[string]Absence         `json:"actions"`
        Checklist map[string][]ChecklistItem `json:"checklist"`
        Seq       map[string]int64           `json:"seq"`
        Gaps      map[string]map[string]*Gap `json:"gaps"`
        Sessions  map[string]Session         `json:"sessions"`
        Messages  map[string]Message         `json:"messages"`
        BySess    map[string][]string        `json:"by_session"`
}

const snapshotVersion = 1

// Persister сохраняет оба хранилища в один JSON-файл и восстанавливает их при старте.
// Это замена БД для прототипа: перезапуск сервера не теряет заявки, чек-листы и диалоги.
type Persister struct {
        Path  string
        Store *Store
        Msgs  *MessageStore
        saved int64
}

func (p *Persister) version() int64 { return p.Store.Version() + p.Msgs.Version() }

// Load читает снимок. Отсутствие файла — не ошибка (первый запуск).
func (p *Persister) Load() error {
        raw, err := os.ReadFile(p.Path)
        if errors.Is(err, os.ErrNotExist) {
                return nil
        }
        if err != nil {
                return err
        }
        var snap snapshot
        if err := json.Unmarshal(raw, &snap); err != nil {
                return err
        }
        if snap.V != snapshotVersion {
                return errors.New("unsupported snapshot version")
        }

        s := p.Store
        s.mu.Lock()
        if snap.Actions != nil {
                s.absences = snap.Actions
        }
        if snap.Checklist != nil {
                s.checklist = snap.Checklist
        }
        if snap.Seq != nil {
                s.seq = snap.Seq
        }
        if snap.Gaps != nil {
                s.gaps = snap.Gaps
        }
        s.mu.Unlock()

        m := p.Msgs
        m.mu.Lock()
        if snap.Sessions != nil {
                m.sessions = snap.Sessions
        }
        if snap.Messages != nil {
                m.messages = snap.Messages
        }
        if snap.BySess != nil {
                m.bySess = snap.BySess
        }
        // Ответы, которые не успели завершиться до остановки, уже никто не досчитает.
        for id, msg := range m.messages {
                if msg.Role == "assistant" && (msg.Status == "pending" || msg.Status == "in_progress") {
                        msg.Status = "failed"
                        msg.Content = "Сервер был перезапущен, отправьте вопрос ещё раз."
                        m.messages[id] = msg
                }
        }
        m.mu.Unlock()

        p.saved = p.version()
        return nil
}

// Save пишет снимок атомарно: во временный файл и переименованием.
func (p *Persister) Save() error {
        ver := p.version()
        s, m := p.Store, p.Msgs
        s.mu.RLock()
        m.mu.RLock()
        raw, err := json.Marshal(snapshot{
                V: snapshotVersion, SavedAt: time.Now(),
                Actions: s.absences, Checklist: s.checklist, Seq: s.seq, Gaps: s.gaps,
                Sessions: m.sessions, Messages: m.messages, BySess: m.bySess,
        })
        m.mu.RUnlock()
        s.mu.RUnlock()
        if err != nil {
                return err
        }
        if err := os.MkdirAll(filepath.Dir(p.Path), 0o755); err != nil {
                return err
        }
        tmp := p.Path + ".tmp"
        if err := os.WriteFile(tmp, raw, 0o600); err != nil {
                return err
        }
        if err := os.Rename(tmp, p.Path); err != nil {
                return err
        }
        p.saved = ver
        return nil
}

// Run сохраняет состояние раз в interval, если оно изменилось, сразу — по сигналу Store.RequestSave
// (подтверждённая заявка не теряется при падении между тиками) и ещё раз — при остановке.
func (p *Persister) Run(ctx context.Context, interval time.Duration) {
        t := time.NewTicker(interval)
        defer t.Stop()
        for {
                select {
                case <-ctx.Done():
                        if err := p.Save(); err != nil {
                                slog.Error("state save on shutdown failed", "err", err)
                        }
                        return
                case <-t.C:
                        p.saveIfChanged()
                case <-p.Store.SaveRequests():
                        p.saveIfChanged()
                }
        }
}

func (p *Persister) saveIfChanged() {
        if p.version() == p.saved {
                return
        }
        if err := p.Save(); err != nil {
                slog.Error("state save failed", "err", err)
        }
}

// Типы действий, которые агент готовит как черновик и пользователь подтверждает.
const (
        ActionAbsence = "absence"
        ActionAccess  = "access"
)

// Статусы действия: draft → pending → approved | rejected; draft | pending → cancelled.
const (
        StatusDraft     = "draft"
        StatusPending   = "pending"
        StatusApproved  = "approved"
        StatusRejected  = "rejected"
        StatusCancelled = "cancelled"
)

// Статусы шага чек-листа.
const (
        StepTodo    = "todo"
        StepWaiting = "waiting" // ждёт внешнего действия (например, выдачи доступа)
        StepDone    = "done"
)

// Event — запись истории действия: кто и когда перевёл его в статус.
type Event struct {
        At      time.Time `json:"at"`
        By      string    `json:"by"`
        Status  string    `json:"status"`
        Comment string    `json:"comment,omitempty"`
}

// Absence — запись действия пользователя (исторически только отгул; теперь также запрос доступа).
// Имя типа сохранено ради обратной совместимости JSON и тестов.
type Absence struct {
        ID        string    `json:"id"`
        Type      string    `json:"type"`               // absence | access
        Resource  string    `json:"resource,omitempty"` // для access: jira, gitlab, grafana…
        UserID    string    `json:"user_id"`
        TeamID    string    `json:"team_id,omitempty"`
        DateFrom  string    `json:"date_from"`
        DateTo    string    `json:"date_to"`
        Reason    string    `json:"reason"`
        Status    string    `json:"status"`
        CreatedAt time.Time `json:"created_at"`
        UpdatedAt time.Time `json:"updated_at,omitempty"`
        History   []Event   `json:"history,omitempty"`
}

type ChecklistItem struct {
        ID       string `json:"id"`
        Title    string `json:"title"`
        Stage    string `json:"stage"`
        Done     bool   `json:"done"`
        Status   string `json:"status"`                      // todo | waiting | done
        Kind     string `json:"kind,omitempty"`              // access — шаг связан с заявками на доступ
        ActionID string `json:"related_action_id,omitempty"` // заявка, которой ждёт шаг
        Owner    string `json:"owner"`
}

func (it *ChecklistItem) normalize() {
        switch {
        case it.Done:
                it.Status = StepDone
        case it.Status == "" || it.Status == StepDone:
                it.Status = StepTodo
        }
}

type Store struct {
        mu        sync.RWMutex
        absences  map[string]Absence
        checklist map[string][]ChecklistItem
        seq       map[string]int64
        gaps      map[string]map[string]*Gap // team → нормализованный вопрос → счётчик
        version   atomic.Int64               // растёт при каждом изменении (для сохранения на диск)
        now       func() time.Time
        saveNow   chan struct{} // сигнал Persister'у «сохранить сейчас» (буфер 1)
}

func New() *Store {
        return &Store{
                absences:  map[string]Absence{},
                checklist: map[string][]ChecklistItem{},
                seq:       map[string]int64{},
                gaps:      map[string]map[string]*Gap{},
                now:       time.Now,
                saveNow:   make(chan struct{}, 1),
        }
}

// Version — счётчик изменений; по нему Persister понимает, что пора сохранять.
func (s *Store) Version() int64 { return s.version.Load() }

// RequestSave просит Persister сохранить состояние сейчас, не дожидаясь тика (после подтверждения,
// согласования, отклонения или отмены заявки). Не блокирует: повторные сигналы до сохранения схлопываются.
// Без Persister (STATE_FILE пуст) сигнал просто остаётся в буфере — поведение не меняется.
func (s *Store) RequestSave() {
        select {
        case s.saveNow <- struct{}{}:
        default:
        }
}

// SaveRequests — канал сигналов «сохранить сейчас» для Persister.
func (s *Store) SaveRequests() <-chan struct{} { return s.saveNow }

func (s *Store) touch() { s.version.Add(1) }

// NextID возвращает следующий ID с префиксом, например ABS-000001 или REQ-000001.
func (s *Store) NextID(prefix string) string {
        s.mu.Lock()
        defer s.mu.Unlock()
        s.seq[prefix]++
        s.touch()
        return fmt.Sprintf("%s-%06d", prefix, s.seq[prefix])
}

// NextAbsenceID оставлен для совместимости.
func (s *Store) NextAbsenceID() string { return s.NextID("ABS") }

func (s *Store) CreateAbsence(a Absence) Absence {
        a = s.prepare(a)
        s.mu.Lock()
        defer s.mu.Unlock()
        s.absences[a.ID] = a
        s.touch()
        return cloneAction(a)
}

// prepare заполняет значения по умолчанию: тип, время создания и первую запись истории.
func (s *Store) prepare(a Absence) Absence {
        if a.Type == "" {
                a.Type = ActionAbsence
        }
        if a.CreatedAt.IsZero() {
                a.CreatedAt = s.now()
        }
        if a.UpdatedAt.IsZero() {
                a.UpdatedAt = a.CreatedAt
        }
        if len(a.History) == 0 && a.Status != "" {
                a.History = []Event{{At: a.CreatedAt, By: a.UserID, Status: a.Status}}
        }
        return a
}

func cloneAction(a Absence) Absence {
        if a.History != nil {
                a.History = append([]Event(nil), a.History...)
        }
        return a
}

func (s *Store) GetAbsence(id string) (Absence, bool) {
        s.mu.RLock()
        defer s.mu.RUnlock()
        a, ok := s.absences[id]
        return cloneAction(a), ok
}

func sortActions(out []Absence) {
        sort.Slice(out, func(i, j int) bool {
                if out[i].CreatedAt.Equal(out[j].CreatedAt) {
                        return out[i].ID > out[j].ID
                }
                return out[i].CreatedAt.After(out[j].CreatedAt)
        })
}

// ListByUser возвращает действия пользователя, новые сверху.
func (s *Store) ListByUser(userID string) []Absence {
        s.mu.RLock()
        out := make([]Absence, 0)
        for _, a := range s.absences {
                if a.UserID == userID {
                        out = append(out, cloneAction(a))
                }
        }
        s.mu.RUnlock()
        sortActions(out)
        return out
}

// ListByTeam — действия команды в заданном статусе (пустой статус — все, кроме черновиков).
func (s *Store) ListByTeam(teamID, status string) []Absence {
        s.mu.RLock()
        out := make([]Absence, 0)
        for _, a := range s.absences {
                if a.TeamID != teamID {
                        continue
                }
                if (status == "" && a.Status != StatusDraft) || a.Status == status {
                        out = append(out, cloneAction(a))
                }
        }
        s.mu.RUnlock()
        sortActions(out)
        return out
}

// Transition атомарно переводит действие из статуса from в to (защита от двойного подтверждения).
func (s *Store) Transition(id, from, to string) (Absence, bool) {
        return s.TransitionBy(id, []string{from}, to, "", "")
}

// TransitionBy — атомарный переход из любого из статусов from в to с записью в историю.
func (s *Store) TransitionBy(id string, from []string, to, by, comment string) (Absence, bool) {
        s.mu.Lock()
        defer s.mu.Unlock()
        a, ok := s.absences[id]
        if !ok {
                return Absence{}, false
        }
        allowed := false
        for _, f := range from {
                if a.Status == f {
                        allowed = true
                        break
                }
        }
        if !allowed {
                return Absence{}, false
        }
        if by == "" {
                by = a.UserID
        }
        a.Status = to
        a.UpdatedAt = s.now()
        a.History = append(append([]Event(nil), a.History...), Event{At: a.UpdatedAt, By: by, Status: to, Comment: comment})
        s.absences[id] = a
        s.touch()
        return cloneAction(a), true
}

func (s *Store) SeedChecklist(userID string, items []ChecklistItem) {
        s.mu.Lock()
        defer s.mu.Unlock()
        cp := make([]ChecklistItem, len(items))
        copy(cp, items)
        for i := range cp {
                cp[i].Owner = userID
                cp[i].normalize()
        }
        s.checklist[userID] = cp
        s.touch()
}

// HasChecklist — есть ли у пользователя чек-лист (чтобы не затирать загруженное с диска сидом).
func (s *Store) HasChecklist(userID string) bool {
        s.mu.RLock()
        defer s.mu.RUnlock()
        return len(s.checklist[userID]) > 0
}

// Checklist возвращает копию, чтобы вызывающий код не читал слайс без блокировки.
func (s *Store) Checklist(userID string) []ChecklistItem {
        s.mu.RLock()
        defer s.mu.RUnlock()
        items := s.checklist[userID]
        out := make([]ChecklistItem, len(items))
        copy(out, items)
        return out
}

// MarkDoneOwned отмечает шаг владельца выполненным. Шаг, связанный с заявками на доступ
// (kind=access) или ждущий решения по заявке (waiting), вручную не закрывается (locked):
// он становится done только после одобрения заявки руководителем.
func (s *Store) MarkDoneOwned(userID, itemID string) (found, owned, locked bool) {
        s.mu.Lock()
        defer s.mu.Unlock()
        for uid, items := range s.checklist {
                for i := range items {
                        if items[i].ID == itemID {
                                if uid != userID {
                                        return true, false, false
                                }
                                if items[i].Done {
                                        return true, true, false
                                }
                                if items[i].Status == StepWaiting || items[i].Kind == ActionAccess {
                                        return true, true, true
                                }
                                items[i].Done = true
                                items[i].Status = StepDone
                                s.touch()
                                return true, true, false
                        }
                }
        }
        return false, false, false
}

func (s *Store) MarkDone(userID, itemID string) bool {
        _, owned, locked := s.MarkDoneOwned(userID, itemID)
        return owned && !locked
}

// LinkStep связывает первый незавершённый шаг пользователя вида kind с заявкой и ставит ему статус.
// Так шаг «Получить доступы» переходит в waiting после отправки заявки и в done после её одобрения.
func (s *Store) LinkStep(userID, kind, actionID, status string) (ChecklistItem, bool) {
        s.mu.Lock()
        defer s.mu.Unlock()
        items := s.checklist[userID]
        for i := range items {
                if items[i].Kind != kind || items[i].Done {
                        continue
                }
                items[i].ActionID = actionID
                items[i].Status = status
                items[i].Done = status == StepDone
                s.touch()
                return items[i], true
        }
        return ChecklistItem{}, false
}

// UnlinkStep возвращает шаг, ждавший заявку actionID, в todo (заявку отклонили или отменили).
func (s *Store) UnlinkStep(userID, actionID string) {
        s.mu.Lock()
        defer s.mu.Unlock()
        items := s.checklist[userID]
        for i := range items {
                if items[i].ActionID == actionID && !items[i].Done {
                        items[i].ActionID = ""
                        items[i].Status = StepTodo
                        s.touch()
                }
        }
}

// Ограничения действий едины для всех входов: POST /actions, чат, MCP — и повторно при подтверждении.
const (
        MaxReasonRunes = 500
        MaxAbsenceDays = 30
        // MaxOpenDrafts — сколько неподтверждённых черновиков может держать один пользователь.
        MaxOpenDrafts = 50
)

// ErrTooManyDrafts — у пользователя слишком много неподтверждённых черновиков.
var ErrTooManyDrafts = errors.New("too many drafts")

// accessResources — системы, доступ к которым можно запросить (whitelist).
var accessResources = map[string]bool{"jira": true, "gitlab": true, "grafana": true, "git": true}

// NormalizeResource приводит имя системы к виду из whitelist («GitLab » → «gitlab»).
func NormalizeResource(r string) string {
        return strings.ToLower(strings.TrimSpace(r))
}

// ValidResource — можно ли запросить доступ к этой системе.
func ValidResource(r string) bool { return accessResources[r] }

// ValidateAbsenceDates: формат YYYY-MM-DD, конец не раньше начала, начало не в прошлом, не дольше 30 дней.
func ValidateAbsenceDates(from, to string, now time.Time) error {
        f, err1 := time.Parse("2006-01-02", from)
        t, err2 := time.Parse("2006-01-02", to)
        switch {
        case err1 != nil || err2 != nil:
                return errors.New("date_from/date_to must be YYYY-MM-DD")
        case t.Before(f):
                return errors.New("date_to must be >= date_from")
        case from < now.Format("2006-01-02"):
                return errors.New("date_from is in the past")
        case t.Sub(f) > MaxAbsenceDays*24*time.Hour:
                return errors.New("absence longer than 30 days is not supported")
        }
        return nil
}

// ValidateAction проверяет действие перед созданием черновика и перед подтверждением.
func ValidateAction(a Absence, now time.Time) error {
        if utf8.RuneCountInString(a.Reason) > MaxReasonRunes {
                return errors.New("reason too long (max 500 chars)")
        }
        switch a.Type {
        case ActionAbsence:
                return ValidateAbsenceDates(a.DateFrom, a.DateTo, now)
        case ActionAccess:
                if a.Resource == "" {
                        return errors.New("resource required")
                }
                if !ValidResource(a.Resource) {
                        return errors.New("resource must be one of: jira, gitlab, grafana, git")
                }
                return nil
        }
        return errors.New("type must be absence or access")
}

// CreateDraft атомарно проверяет лимит открытых черновиков пользователя и сохраняет черновик.
func (s *Store) CreateDraft(a Absence) (Absence, error) {
        a.Status = StatusDraft
        a = s.prepare(a)
        s.mu.Lock()
        defer s.mu.Unlock()
        n := 0
        for _, x := range s.absences {
                if x.UserID == a.UserID && x.Status == StatusDraft {
                        n++
                }
        }
        if n >= MaxOpenDrafts {
                return Absence{}, ErrTooManyDrafts
        }
        s.absences[a.ID] = a
        s.touch()
        return cloneAction(a), nil
}

// Spec — описание инструмента для MCP (tools/list) и для промпта LLM.
type Spec struct {
        Name                 string         `json:"name"`
        Description          string         `json:"description"`
        Permission           string         `json:"permission"`
        RequiresConfirmation bool           `json:"requiresConfirmation"`
        InputSchema          map[string]any `json:"inputSchema"`
}

func object(required []string, props map[string]any) map[string]any {
        s := map[string]any{"type": "object", "properties": props}
        if len(required) > 0 {
                s["required"] = required
        }
        return s
}

func str(desc string) map[string]any { return map[string]any{"type": "string", "description": desc} }

func date(desc string) map[string]any {
        return map[string]any{"type": "string", "format": "date", "description": desc}
}

// specs — единый источник описаний; синхронизирован со specs/contracts/mcp-tools.json (проверяется тестом).

                        "resource": map[string]any{"type": "string", "enum": []string{"jira", "gitlab", "grafana", "git"}},
                        "reason":   str("зачем нужен доступ"),
                }),
        },
        "directory_find_owner": {
                Description: "Кто отвечает за тему (доступы, инструменты, чек-лист, отгулы)",
                InputSchema: object([]string{"query"}, map[string]any{"query": str("тема")}),
        },
}

// Describe возвращает описание зарегистрированного инструмента.
func Describe(t Tool) Spec {
        s := specs[t.Name()]
        s.Name = t.Name()
        s.Permission = t.RequiredPermission()
        s.RequiresConfirmation = t.RequiresConfirmation()
        if s.InputSchema == nil {
                s.InputSchema = object(nil, map[string]any{})
        }
        return s
}

// List — описания всех зарегистрированных инструментов по алфавиту.
func (r *Registry) List() []Spec {
        out := make([]Spec, 0, len(r.tools))
        for _, t := range r.tools {
                out = append(out, Describe(t))
        }
        sort.Slice(out, func(i, j int) bool { return out[i].Name < out[j].Name })
        return out
}

import (

type Result struct {
        OK      bool           `json:"ok"`
        Message string         `json:"message"`
        Data    map[string]any `json:"data,omitempty"`
}

type Tool interface {
        Name() string
        RequiredPermission() string
        RequiresConfirmation() bool
        Run(ctx context.Context, user auth.User, args map[string]string) (Result, error)
}

type Registry struct {
        tools map[string]Tool
}

func NewRegistry() *Registry { return &Registry{tools: map[string]Tool{}} }

func (r *Registry) Register(t Tool) { r.tools[t.Name()] = t }

func (r *Registry) Get(name string) (Tool, bool) {
        t, ok := r.tools[name]
        return t, ok
}

// Run выполняет инструмент с таймаутом 10с.
func Run(ctx context.Context, t Tool, u auth.User, args map[string]string) (Result, error) {
        ctx, cancel := context.WithTimeout(ctx, 10*time.Second)
        defer cancel()
        return t.Run(ctx, u, args)
}
